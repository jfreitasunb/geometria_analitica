\chapter{Quádricas} % (fold)
\label{cha:quadricas}

Em $\real^3$ estamos interessados em objetos definidos por uma equação de segundo grau em $x$, $y$ e $z$ da forma
\begin{equation}\label{equacao_quadrica}
    ax^2 + by^2 + cz^2 + dxy + exz + fyz + gx + hy + iz + j = 0.
\end{equation}

O conjunto de pontos em $\real^3$ satisfazendo uma equação da forma \eqref{equacao_quadrica} é chamado de uma \textbf{quádrica} ou \textbf{superfície quádrica}. Observe que pelo menos um dos números $a$, $b$, $c$, $d$, $e$ ou $f$ deve ser não nulo.\index{Quádricas}

Nosso principal objetivo será tentar simplificar a equação \eqref{equacao_quadrica} de modo que seja possível identificar que objeto tal equação representa.

\section{Esfera}

\begin{definicao}
    Uma \textbf{esfera} é uma quádrica com equação na forma\index{Quádricas!Esfera}
    \begin{equation}\label{equacao_esfera}
        x^2 + y^2 + z^2 = r^2.
    \end{equation}
\end{definicao}

\end{figure}

\section{Elipsoide}

\begin{definicao}
    Um \textbf{elipsoide} é uma quádrica descrita pela equação
    \begin{equation}\label{equacao_elipsoide}
        \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} + \dfrac{z^2}{c^2} = 1
    \end{equation}
    onde os números reais $a$, $b$ e $c$ são positivos e pelo menos dois deles são distintos.\index{Quádricas!Elipsoide}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Elipsoide: $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} + \dfrac{z^2}{c^2} = 1$}
    \includegraphics[scale=0.2]{elipsoide.png}
\end{figure}

Considere o plano $\pi: z = k$. Qual a interseção de $\pi$ com o elipsoide de equação \eqref{equacao_elipsoide}? A interseção é dada pelo sistema
\[
    \begin{cases}
        \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} + \dfrac{z^2}{c^2} = 1 \\
        z = k,
    \end{cases}
\]
isto é, pela equação
\begin{equation}\label{intersecao_elipsoide_planoz}
    \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1 - \dfrac{k^2}{c^2}.
\end{equation}

Se $k^2 > c^2$, então a equação \eqref{intersecao_elipsoide_planoz} não admite solução pois $1 - k^2/c^2 < 0$. Assim seja $k^2 \le c^2$, isto é, $-c \le k \le c$. Se $k = c$, então a interseção é o ponto $(0,0,c)$ e se $k = -c$ é o ponto $(0,0,-c)$. Então seja $-c < k < c$ e denote $p = 1 - k^2/c^2$. Podemos reescrever \eqref{intersecao_elipsoide_planoz} como
\begin{equation}\label{resultado_intersecao_elipsoide_planoz}
    \dfrac{x^2}{pa^2} + \dfrac{y^2}{pb^2} = 1.
\end{equation}

Se $a = b$, então a equação \eqref{resultado_intersecao_elipsoide_planoz} é uma circunferência. Se $a \ne b$, então a equação \eqref{resultado_intersecao_elipsoide_planoz} descreve uma elipse.

De modo análogo, se tomarmos $\pi : x = k$ ou $\pi : y = k$, as interseções serão sempre ou conjunto vazio ou um ponto ou uma circunferência ou uma elipse.

Na equação \eqref{equacao_elipsoide} se $z = 0$, a equação torna-se
\[
    \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1
\]
e a superfície resultante é chamada de \textbf{cilindro elíptico.}\index{Quádricas!Cilindro Elíptico}

\begin{figure}[H]
    \centering
    \caption{Cilindro Elíptico: $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$}
    \includegraphics[scale=0.35]{cilindro_eliptico.png}
\end{figure}

\section{Paraboloides}

\begin{definicao}
    Sejam $a$ e $b$ números reais positivos a quádrica descrita de equação
    \[
        z = \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2},
    \]
    é chamada de:
    \begin{enumerate}
        \item um \textbf{paraboloide elíptico} se $a \ne b$;\index{Quádricas!Parabolóide elíptico}
        \item um \textbf{paraboloide de rotação} se $a = b$;\index{Quádricas!Parabolóide de rotação}
    \end{enumerate}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Paraboloide Elíptico: $z = \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2}$}
    \includegraphics[scale=0.2]{paraboloide_eliptico.png}
\end{figure}

\begin{figure}[H]
    \centering
    \caption{Paraboloide Rotação: $z = \dfrac{x^2}{a^2} + \dfrac{y^2}{a^2}$}
    \includegraphics[scale=0.2]{paraboloide_rotacao.png}
\end{figure}

A interseção do paraboloide
\[
    z = \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2},
\]
com o plano $\pi: x = k$ é dada por
\[
    z = \dfrac{k^2}{a^2} + \dfrac{y^2}{b^2},
\]
que é uma parábola. O mesmo ocorrerá com o plano $\pi: y = k$.

Agora a interseção com $z = k$ é dada forma
\[
    \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = k.
\]

Se $k < 0$, então a interseção é vazia. Se $k = 0$, a interseção é o ponto $(0,0,0)$. Se $k > 0$, precisamos analisar o que ocorre com $a$ e $b$: se $a \ne b$, então obtemos uma elipse. Se $a = b$, então a interseção é uma circunferência.

\begin{definicao}
    Sejam $a$ e $b$ números reais positivos. A quádrica descrita pela equação
    \[
        z = \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2},
    \]
    é chamada de um um \textbf{paraboloide hiperbólico}.\index{Quádricas!Parabolóide hiperbólico}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Paraboloide Hiperbólico: $z = \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2}$}
    \includegraphics[scale=0.2]{paraboloide_hiperbolico.png}
\end{figure}

A interseção do paraboloide hiperbólico com o plano $\pi : z = k$ é dada por
\[
    \begin{cases}
        \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2} = k \\
        z = k
    \end{cases}.
\]

Se $k = 0$, então obtemos
\[
    \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2} = \left(\dfrac{y}{b} - \dfrac{x}{a}\right)\left(\dfrac{y}{b} + \dfrac{x}{a}\right) = 0
\]
que trata-se de duas retas concorrentes na origem:
\[
    r : \begin{cases}
        \dfrac{y}{b} - \dfrac{x}{a} = 0 \\
        z = 0
    \end{cases}, \quad s : \begin{cases}
        \dfrac{y}{b} - \dfrac{x}{a} = 0 \\
        z = 0
    \end{cases}.
\]

Se $k \ne 0$, podemos escrever
\[
    \dfrac{y^2}{kb^2} - \dfrac{x^2}{ka^2} = 1
\]
que é uma hipérbole.

A interseção com o plano $\pi : y = k$ é dada por
\[
    z = \dfrac{k^2}{b^2} - \dfrac{x^2}{a^2}
\]
que é uma parábola. O mesmo ocorre com o plano $\pi : x = k$.

\begin{definicao}
    Um \textbf{cilindro parabólico} é uma quádrica descrita pela equação
    \[
        y = ax^2
    \]
    onde $a$ é um número real.\index{Quádricas!Cilindro Parabólico}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Cilindro parabólico: $y = ax^2$}
    \includegraphics[scale=0.2]{cilindro_parabolico.png}
\end{figure}

\section{Hiperboloides}

\begin{definicao}
    Um \textbf{hiperboloide de uma folha} é uma quádrica descrita pela equação
    \[
        \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} - \dfrac{z^2}{c^2} = 1
    \]
    onde os números reais $a$, $b$ e $c$ são positivos e pelo menos dois deles são distintos.\index{Quádricas!Hiperboloide de uma folha}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Hiperboloide de uma folha: $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} - \dfrac{z^2}{c^2} = 1$}
    \includegraphics[scale=0.2]{hiperboloide_de_uma_folha.png}
\end{figure}

Para o caso do hiperboloide de uma folha a interseção com o plano $\pi : z = k$ é dada pelo sistema
\[
    \begin{cases}
        \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1 + \dfrac{k^2}{c^2} \\
        z = k.
    \end{cases}
\]

Denotando $p = 1 + k^2/c^2 > 0$ podemos escrever
\begin{equation}\label{resultado_intersecao_hiperboloide_1_folha_planoz}
    \dfrac{x^2}{pa^2} + \dfrac{y^2}{pb^2} = 1.
\end{equation}

Se $a = b$, então a equação \eqref{resultado_intersecao_hiperboloide_1_folha_planoz} descreve uma circunferência. Se $a \ne b$, então a equação \eqref{resultado_intersecao_hiperboloide_1_folha_planoz} descreve uma elipse.

Agora a interseção com o plano $\pi : y = k$ é descrita pela equação
\begin{equation}\label{intersecao_hiperboloide_1_folha_planoy}
    \dfrac{x^2}{a^2} - \dfrac{z^2}{c^2} = 1 - \dfrac{k^2}{b^2}.
\end{equation}
Se $k^2 = b^2$, obtemos
\[
    \dfrac{x^2}{a^2} - \dfrac{z^2}{c^2} = 0,
\]
isto é,
\[
    \left(\dfrac{x}{a} - \dfrac{z}{c}\right)\left(\dfrac{x}{a} + \dfrac{z}{c}\right).
\]
Logo a interseção de $\pi$ com o hiperboloide de uma folha é o par de retas concorrentes
\[
    r : \begin{cases}
        \dfrac{x}{a} - \dfrac{z}{c} = 0 \\
        y = k
    \end{cases}, \quad s : \begin{cases}
        \dfrac{x}{a} + \dfrac{z}{c} = 0 \\
        y = k
    \end{cases}.
\]

Se $k^2 \ne b^2$, então fazendo $p = 1 - k^2/b^2$ podemos escrever a equação \eqref{intersecao_hiperboloide_1_folha_planoy} como
\[
    \dfrac{x^2}{pa^2} - \dfrac{z^2}{pc^2} = 1
\]
obtendo-se uma hipérbole.

De modo análogo, obtemos as mesmas interseções se considerarmos o plano $\pi : x = k$.

\begin{definicao}
    Um \textbf{hiperboloide de duas folhas} é uma quádrica descrita pela equação
    \[
        \dfrac{z^2}{c^2} - \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2} = 1
    \]
    onde os números reais $a$, $b$ e $c$ são positivos e pelo menos dois deles são distintos.\index{Quádricas!Hiperboloide de duas folhas}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Hiperboloide duas folhas: $\dfrac{z^2}{c^2} - \dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$}
    \includegraphics[scale=0.2]{hiperboloide_de_duas_folhas.png}
\end{figure}

A interseção do hiperboloide de duas folhas com a plano $\pi: y = k$ é dada por
\[
    \dfrac{z^2}{c^2} - \dfrac{x^2}{a^2} = 1 +  \dfrac{k^2}{b^2}.
\]
Denotando $p = 1 + k^2/b^2$ podemos escreve a equação anterior como
\[
    \dfrac{z^2}{pc^2} - \dfrac{x^2}{pa^2} = 1
\]
que representa uma hipérbole.

A interseção com o plano $\pi: x = k$ é dada por
\[
    \dfrac{z^2}{c^2} - \dfrac{y^2}{b^2} = 1 +  \dfrac{k^2}{a^2}.
\]
e fazendo $p = 1 + k^2/a^2$ obtemos
\[
    \dfrac{z^2}{pc^2} - \dfrac{y^2}{pb^2} = 1
\]
que também é uma hipérbole.

Agora a interseção com o plano $\pi : z = k$ é dada por
\begin{equation}\label{intersecao_hiperboloide_2_folhas_planoy}
    \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = \dfrac{k^2}{c^2} - 1.
\end{equation}

Se $k^2 < c^2$, então $k^2/c^2 - 1 < 0$ e daí a equação \eqref{intersecao_hiperboloide_2_folhas_planoy} não admite solução.

Se $k^2 = c^2$, então a interseção é ou o ponto $(0,k,0)$ ou o ponto $(0,-k,0)$.

Se $k^2 > c^2$, isto é, $k < -c$ ou $k > c$, então tomando $p = k^2/c^2 - 1 > 0$, podemos escrever \eqref{intersecao_hiperboloide_2_folhas_planoy} como
\[
    \dfrac{x^2}{pa^2} + \dfrac{y^2}{pb^2} = 1.
\]
Assim se $a = b$, então obtemos uma circunferência e se $a \ne b$, então a interseção é uma elipse.

\begin{definicao}
    Um \textbf{cilindro hiperbólico} é uma quádrica descrita pela equação
    \[
        \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2} = 1
    \]
    onde os números reais $a$, $b$ e $c$ são positivos e pelo menos dois deles são distintos.\index{Quádricas!Cilindro Hiperbólico}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Cilindro hiperbólico: $\dfrac{y^2}{b^2} - \dfrac{x^2}{a^2} = 1$}
    \includegraphics[scale=0.2]{cilindro_hiperbolico.png}
\end{figure}

\section{Cone Elíptico}

\begin{definicao}
    Um \textbf{cone elíptico} é um conjunto de pontos que satisfaz uma equação da forma
    \begin{equation}\label{cone_eliptico}
        z^2 = \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2}
    \end{equation}
    em que $a$ e $b$ são números reais positivos.\index{Quádricas!Cone Elíptico} Se $a = b$, o cone é chamado de \textbf{cone circular}.\index{Quádricas!Cone Circular}
\end{definicao}

\begin{figure}[H]
    \centering
    \caption{Cone elíptico: $z^2 = \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2}$}
    \includegraphics[scale=0.2]{cone_eliptico.png}
\end{figure}

Seja $k \ne 0$. A interseção do cone elíptico de equação \eqref{cone_eliptico} com o plano $\pi : z = k$ é a elipse de equação
\[
    \dfrac{x^2}{a^2k^2} + \dfrac{y^2}{b^2k^2} = 1.
\]
Já sua interseção com o plano $\pi : y = k$ é a hipérbole de equação
\[
    \dfrac{z^2}{k^2/b^2} - \dfrac{x^2}{a^2k^2/b^2} = 1
\]
e com o plano $\pi : x = k$ é a hipérbole de equação
\[
    \dfrac{z^2}{k^2/a^2} - \dfrac{y^2}{b^2k^2/a^2} = 1.
\]

As equações
\[
    x^2 = \dfrac{y^2}{b^2} + \dfrac{z^2}{c^2}\quad \mbox{e}\quad y^2 = \dfrac{x^2}{a^2} + \dfrac{z^2}{c^2}
\]
também representam cones elípticos.

\section{Identificação de Quádricas}

Considere uma quádrica de equação
\begin{equation}\label{equacao_geral_quadrica}
    ax^2 + by^2 + cz^2 + dxy + exz + fyz + gx + hy + iz + j = 0.
\end{equation}

Queremos remover os termos $xy$, $xz$ e $yz$ da equação \eqref{equacao_geral_quadrica} para então usar uma translação de eixos, se necessário, e com isso identificar a quádrica definida por tal equação. Para isso vamos tentar encontrar uma matriz $Q = [\vec{U_1}\ \vec{U_2}\ \vec{U_3}]$ onde os vetores $\vec{U_1}$, $\vec{U_2}$ e $\vec{U_3}$ são unitários, ortogonais e tais que a mudança
\begin{equation}\label{substituicao_quadrica}
    \begin{bmatrix}
        x \\
        y \\
        z
    \end{bmatrix} = Q \begin{bmatrix}
        x_1 \\
        y_1 \\
        z_1
    \end{bmatrix}
\end{equation}
transforma a equação \eqref{equacao_geral_quadrica} numa equação da forma
\begin{equation}\label{equacao_rotacionada_quadrica}
    a'x_1^2 + b'y_1^2 + c'z_1^2 + g'x_1 + h'y_1 + i'z_1 + j = 0.
\end{equation}

Inicialmente observe que a \eqref{equacao_geral_quadrica} pode ser escrita na forma
\begin{equation}\label{equacao_matricial_quadrica}
    X^tAX + KX + J = [0]
\end{equation}
onde
\[
    A = \begin{bmatrix}
        a   & d/2 & e/2 \\
        d/2 & b   & f/2 \\
        e/2 & f/2 & c
    \end{bmatrix}
    , \quad K = \begin{bmatrix}
        g & h & i
    \end{bmatrix}, \quad X = \begin{bmatrix}
        x \\
        y \\
        z
    \end{bmatrix}\quad\mbox{e}\quad J = [j]
\]
e $t$ denota a matriz transposta. Queremos fazer uma substituição da forma \eqref{substituicao_quadrica}, isto é, $X = QX'$ para com isso obter uma equação na forma
\[
    X_1^tBX_1 + K'X_1 + J = [0]
\]
onde
\[
    B = \begin{bmatrix}
        a'   & d'/2 & e'/2 \\
        d'/2 & b'   & f'/2 \\
        e'/2 & f'/2 & c'
    \end{bmatrix} = Q^tAQ,\quad K' = \begin{bmatrix}
        g' & h' & i'
    \end{bmatrix} = KQ,\quad X = \begin{bmatrix}
        x \\y\\z
    \end{bmatrix},\quad X_1 = \begin{bmatrix}
        x_1 \\y_1\\z_1
    \end{bmatrix}.
\]

Agora observe que a matriz $Q$ é formada por vetores unitários e ortogonais, daí $Q^tQ = I_3$ e com isso obtemos
\begin{align*}
    B - \lambda I_3 = Q^t A Q - \lambda I_3 = Q^t A Q - Q^t\lambda I_3 Q = Q^t (A - \lambda I_3)Q
\end{align*}
para todo $\lambda \in \real$. Com isso
\begin{align*}
    \det(B - \lambda I_3) = \det[Q^t(A - \lambda I_3)Q] = \det(Q^t)\det(A - \lambda I_3)\det(Q) = \det(A - \lambda I_3).
\end{align*}

Assim escolhendo a matriz $Q$ de tal forma que $d' = e' = f' = 0$, o que é sempre possível, obtemos
\begin{align*}
    \det(A - \lambda I_3) = \det(B - \lambda I_3) = \det \begin{bmatrix}
                                                             a' - \lambda & 0            & 0            \\
                                                             0            & b' - \lambda & 0            \\
                                                             0            & 0            & c' - \lambda
                                                         \end{bmatrix} = -(\lambda - a')(\lambda - b')(\lambda - c').
\end{align*}

Logo os coeficientes $a'$, $b'$ e $c'$ são raízes da equação de terceiro grau
\begin{equation}
    p(\lambda) = \det(A - \lambda I_3) = \det \begin{bmatrix}
        a - \lambda & d/2         & e/2         \\
        d/2         & b - \lambda & f/2         \\
        e/2         & f/2         & c - \lambda
    \end{bmatrix}.
\end{equation}

Agora para determinar a matriz $Q$ começamos com a equação
\[
    B = Q^t A Q
\]
e multiplicamos à esquerda por $Q$ obtendo
\[
    Q B = AQ.
\]
Mas
\[
    AQ = A \begin{bmatrix}
        \vec{U_1} & \vec{U_2} & \vec{U_3}
    \end{bmatrix} = \begin{bmatrix}
        A\vec{U_1} & A\vec{U_2} & A\vec{U_3}
    \end{bmatrix}
\]
e
\[
    QB = \begin{bmatrix}
        \vec{U_1} & \vec{U_2} & \vec{U_3}
    \end{bmatrix}\begin{bmatrix}
        a' & 0  & 0  \\
        0  & b' & 0  \\
        0  & 0  & c'
    \end{bmatrix} = \begin{bmatrix}
        a'\vec{U_1} & b'\vec{U_2} & c'\vec{U_3}
    \end{bmatrix}.
\]
Daí $\vec{U_1}$, $\vec{U_2}$ e $\vec{U_3}$ devem satisfazer as equações
\[
    A\vec{U_1} = a'\vec{U_1},\quad A\vec{U_2} = b'\vec{U_2} \quad A\vec{U_3} = c'\vec{U_3}.
\]

Da primeira equação obtemos
\begin{align*}
    A\vec{U_1} = a'I_3\vec{U_1} \\
    (A - a'I_3)\vec{U_1} = \vec{0}.
\end{align*}

Logo $\vec{U_1}$ é uma solução de norma 1 do sistema linear
\[
    (A - a'I_3)X = [0].
\]
Analogamente, $\vec{U_2}$ é uma solução de norma 1 do sistema linear
\[
    (A - b'I_3)X = [0],
\]
que seja ortogonal a $\vec{U_1}$. O terceiro vetor $\vec{U_3}$ é obtido de forma análoga e deve ser ortogonal tanto a $\vec{U_1}$ quanto a $\vec{U_2}$, assim podemos tomar
\[
    \vec{U_3} = \vec{U_1}\times\vec{U_2}.
\]
Como $\vec{U_1}$ e $\vec{U_2}$ são de norma 1, então $\vec{U_3}$ também é de norma 1.

Portanto a mudança que elemina os termos $xy$, $xz$ e $yz$ é dada por $Q = [\vec{U_1}\ \vec{U_2}\ \vec{U_3}]$. Assim temos o seguinte teorema:

\begin{teorema}
    Considere a equação
    \[
        ax^2 + by^2 + cz^2 + dxy + exz + fyz + gx + hy + iz + j = 0
    \]
    com $a$, $b$, $c$, $d$, $e$, $f$, $g$, $h$, $i$, $j \in \real$, sendo $a$, $b$, $c$, $d$, $e$ e $f$ não necessariamente nulos. Então mediante uma mudança da forma
    \[
        X = Q X_1
    \]
    onde
    \[
        X = \begin{bmatrix}
            x \\y\\z
        \end{bmatrix}, \quad
        Q = \begin{bmatrix}
            \vec{U_1} & \vec{U_2} & \vec{U_3}
        \end{bmatrix}, \quad\mbox{e}\quad
        X_1 = \begin{bmatrix}
            x_1 \\y_1\\z_1
        \end{bmatrix}
    \]
    a equação da geral da quádrica acima pode ser transformada em
    \[
        a'x_1^2 + b'y_1^2 + c'z_1^2 + g'x_1 + h'y_1 + i'z_1 + j = 0
    \]
    em que $a'$, $b'$ e $c'$ são raízes de
    \[
        p(\lambda) = \det(A - \lambda I_3) = \det \begin{bmatrix}
            a - \lambda & d/2         & e/2         \\
            d/2         & b - \lambda & f/2         \\
            e/2         & f/2         & c - \lambda
        \end{bmatrix}.
    \]

    Mais ainda, $\vec{U_1}$ é uma solução de norma 1 do sistema linear
    \[
        \begin{bmatrix}
            a - a' & d/2    & e/2    \\
            d/2    & b - a' & f/2    \\
            e/2    & f/2    & c - a'
        \end{bmatrix} \begin{bmatrix}
            x \\y\\z
        \end{bmatrix} = \begin{bmatrix}
            0 \\0\\0
        \end{bmatrix},
    \]
    $\vec{U_2}$ é uma solução de norma 1 do sistema linear
    \[
        \begin{bmatrix}
            a - b' & d/2    & e/2    \\
            d/2    & b - b' & f/2    \\
            e/2    & f/2    & c - b'
        \end{bmatrix} \begin{bmatrix}
            x \\y\\z
        \end{bmatrix} = \begin{bmatrix}
            0 \\0\\0
        \end{bmatrix},
    \]
    e
    \[
        \vec{U_3} = \vec{U_1} \times \vec{U_2}
    \]
\end{teorema}

\begin{teorema}
    Seja $\mathcal{S}$ o conjunto dos pontos do espaço que satisfazem a equação
    \[
        ax^2 + by^2 + cz^2 + dxy + exz + fyz + gx + hy + iz + j = 0,
    \]
    com $a$, $b$, $c$, $d$, $e$, $f$, $g$, $h$, $i$, $j \in \real$, sendo $a$, $b$, $c$, $d$, $e$ e $f$ não simultaneamente nulos. Sejam $a'$, $b'$ e $c'$ raízes de
    \[
        p(\lambda) = \det \begin{bmatrix}
            a - \lambda & d/2         & e/2         \\
            d/2         & b - \lambda & f/2         \\
            e/2         & f/2         & c - \lambda
        \end{bmatrix}.
    \]
    \begin{enumerate}[label=({\roman*})]
        \item Se $a'$, $b'$ e $c'$ tiverem o mesmo sinal, então $\mathcal{S}$ é um elipsoide, um ponto ou o conjunto vazio.
        \item Se $a'$, $b'$ e $c'$ forem não nulos e não tiverem o mesmo sinal, então $\mathcal{S}$ é um hiperboloide de uma folha, de duas folhas ou um cone elíptico.
        \item Se apenas um entre $a'$, $b'$ e $c'$ for nulo, então $\mathcal{S}$ é um paraboloide elíptico, hiperbólico, um cilindro elíptico, hiperbólico, dois planos concorrentes, uma reta ou o conjunto vazio.
        \item Se exatamente dois entre $a'$, $b'$ e $c'$ forem nulos, então $\mathcal{S}$ é um cilindro parabólico, um par de planos paralelos ou um plano.
    \end{enumerate}
\end{teorema}

\begin{exemplos}
    Identifique as seguintes quádricas:
    \begin{enumerate}
        \item $x^2 = 2yz$
              \begin{solucao}
                  Inicialmente essa equação pode ser escrita como
                  \[
                      X^tAX = 0
                  \]
                  onde
                  \[
                      X = \begin{bmatrix}
                          x \\ y\\ z
                      \end{bmatrix},\quad A = \begin{bmatrix}
                          1 & \phantom{-}0 & \phantom{-}0 \\
                          0 & \phantom{-}0 & -1           \\
                          0 & -1           & \phantom{-}0
                      \end{bmatrix}.
                  \]
                  Agora
                  \begin{align*}
                      p(\lambda) & = \det(A - \lambda I_3) = \det \begin{bmatrix}
                                                                      1 - \lambda & \phantom{-}0 & \phantom{-}0 \\
                                                                      0           & -\lambda     & \phantom{-}0 \\
                                                                      0           & -1           & -\lambda
                                                                  \end{bmatrix} \\
                      p(\lambda) & = (1 - \lambda)\lambda^2 - (1 - \lambda)                                 \\
                      p(\lambda) & = (1 - \lambda)(\lambda^2 - 1)
                  \end{align*}
                  cujas raízes são $\lambda = 1$ e $\lambda = \pm 1$. Assim podemos tomar $a' = b' = 1$ e $c' = -1$.

                  Para encontrar $\vec{U_1}$ precisamos resolver o sistema
                  \begin{align*}
                      (A - I_3)X = 0 \\
                      \begin{bmatrix}
                          0 & \phantom{-}0 & \phantom{-}0 \\
                          0 & -1           & -1           \\
                          0 & -1           & -1
                      \end{bmatrix} \begin{bmatrix}
                                        x \\ y\\ z
                                    \end{bmatrix} = \begin{bmatrix}
                                                        0 \\ 0\\ 0
                                                    \end{bmatrix}.
                  \end{align*}
                  Tal sistema possui solução geral dada pelo conjunto
                  \[
                      \{(x, y, z) \mid y + z = 0 \in \real\} = \{(x, -z, z) \mid x, z \in \real\}.
                  \]
                  Agora note que
                  \[
                      (x, -z, z) = x(1, 0, 0) + z(0, 1, 1)
                  \]
                  e os vetores $\vec{v_1} = (1,0,0)$ e $\vec{v_2} = (0,-1,1)$ são ortogonais.

                  Como $a' = b'$ devemos encontrar vetores $\vec{U_1}$ e $\vec{U_2}$ soluções unitárias e ortogonais do sistema linear $(A -I_3)X = 0$. Assim podemos tomar
                  \begin{align*}
                      \vec{U_1} & = \dfrac{1}{\norm{\vec{v_1}}}\vec{v_1} = (1,0,0)                                                 \\
                      \vec{U_2} & = \dfrac{1}{\norm{\vec{v_2}}}\vec{v_2} = \left(0,-\dfrac{1}{\sqrt{2}},\dfrac{1}{\sqrt{2}}\right) \\
                      \vec{U_3} & = \vec{U_1}\times\vec{U_2} = \left(0,-\dfrac{1}{\sqrt{2}},-\dfrac{1}{\sqrt{2}}\right)
                  \end{align*}
                  e com  isso a mudança $X = QX_1$, onde $Q = [\vec{U_1}\ \vec{U_2}\ \vec{U_3}]$ transforma a equação original em
                  \[
                      x_1^2 + y_1^2 - z_1^2 = 0
                  \]
                  que representa um cone circular.
              \end{solucao}

        \item $7x^2 + 10y^2 + 7z^2 - 4xy + 2xz - 4yz - 6 = 0$
              \begin{solucao}
                  Inicialmente essa equação pode ser escrita como
                  \[
                      X^tAX - 6 = 0
                  \]
                  onde
                  \[
                      X = \begin{bmatrix}
                          x \\ y\\ z
                      \end{bmatrix},\quad A = \begin{bmatrix}
                          \phantom{-}7 & -2            & \phantom{-}1 \\
                          -2           & \phantom{-}10 & -2           \\
                          \phantom{-}1 & -2            & \phantom{-}7
                      \end{bmatrix}.
                  \]
                  Agora
                  \begin{align*}
                      p(\lambda) & = \det(A - \lambda I_3) = \det \begin{bmatrix}
                                                                      7 - \lambda  & -2          & \phantom{-}1 \\
                                                                      -2           & 10 -\lambda & -2           \\
                                                                      \phantom{-}1 & -2          & 7 -\lambda
                                                                  \end{bmatrix}                   \\
                      p(\lambda) & = (7 - \lambda)^2(10 - \lambda) + 4 + 4 - (10 - \lambda) - 4(7 - \lambda) - 4(7 - \lambda) \\
                      p(\lambda) & = (10 - \lambda)[(7 - \lambda)^2 - 1)] - 8(6 - \lambda)                                    \\
                      p(\lambda) & = (10 - \lambda)(6 - \lambda)(8 - \lambda) - 8(6 - \lambda)                                \\
                      p(\lambda) & = (6 - \lambda)(\lambda^2 - 18\lambda + 72)                                                \\
                      p(\lambda) & = (6 - \lambda)^2(12 - \lambda)
                  \end{align*}
                  cujas raízes são $\lambda = 6$ e $\lambda = 12$. Assim podemos tomar $a' = b' = 6$ e $c' = 12$.

                  Para encontrar $\vec{U_1}$ precisamos resolver o sistema
                  \begin{align*}
                      (A - 6I_3)X                                   & = 0 \\
                      \begin{bmatrix}
                          \phantom{-}1 & -2           & \phantom{-}1 \\
                          -2           & \phantom{-}4 & -2           \\
                          \phantom{-}1 & -2           & \phantom{-}1
                      \end{bmatrix} \begin{bmatrix}
                                        x \\ y\\ z
                                    \end{bmatrix} & = \begin{bmatrix}
                                                          0 \\ 0\\ 0
                                                      \end{bmatrix}.
                  \end{align*}
                  Tal sistema possui solução geral dada pelo conjunto
                  \[
                      \{(x, y, z) \mid x + 2y + z = 0 \in \real\} = \{(2y - z, y, z) \mid x, z \in \real\}.
                  \]
                  Agora note que
                  \[
                      (2y - z, y, z) = z(-1, 0, 1) + y(2, 1, 0)
                  \]
                  e observe que neste caso os vetores $\vec{v_1} = (-1,0,1)$ e $\vec{v_2} = (2,1,0)$ não são ortogonais. Tomando
                  \[
                      \vec{w} = \vec{v_2} - \proj_{\vec{v_1}}\vec{v_2} = (1, 1, 1)
                  \]
                  então $\vec{w}\perp \vec{v_1}$.

                  Como $a' = b'$ devemos encontrar vetores $\vec{U_1}$ e $\vec{U_2}$ soluções unitárias e ortogonais do sistema linear $(A - 6I_3)X = 0$. Assim podemos tomar
                  \begin{align*}
                      \vec{U_1} & = \dfrac{1}{\norm{\vec{v_1}}}\vec{v_1} = \left(-\dfrac{1}{\sqrt{2}},0,\dfrac{1}{\sqrt{2}}\right)              \\
                      \vec{U_2} & = \dfrac{1}{\norm{\vec{w}}}\vec{w} = \left(\dfrac{1}{\sqrt{3}},\dfrac{1}{\sqrt{3}},\dfrac{1}{\sqrt{3}}\right) \\
                      \vec{U_3} & = \vec{U_1}\times\vec{U_2} = \left(-\dfrac{1}{\sqrt{6}},\dfrac{2}{\sqrt{6}},-\dfrac{1}{\sqrt{6}}\right)
                  \end{align*}
                  e com isso a mudança $X = QX_1$, onde $Q = [\vec{U_1}\ \vec{U_2}\ \vec{U_3}]$ transforma a equação original em
                  \begin{align*}
                      6x_1^2 + 6y_1^2 + 12z_1^2 = 6 \\
                      x_1^2 + y_1^2 + \dfrac{z_1^2}{1/2} = 1
                  \end{align*}
                  que representa um elipsoide.
              \end{solucao}
    \end{enumerate}
\end{exemplos}
