    \questao Sejam $\vec{u} = (x_1, y_1)$ um vetor em $\real^2$ e $\alpha \in \real$. Prove que:
        \begin{itemize}
            \item[(S1)] $(-\alpha)\vec{u} = -(\alpha\vec{u})$;
            \item[(S2)] $\alpha(-\vec{u}) = -(\alpha\vec{u})$;
            \item[(S3)] $(-\alpha)(-\vec{u}) = \alpha\vec{u}$.
        \end{itemize}


    \questao{} Prove que o vetor nulo é único.


    \questao{} Prove que cada vetor $\vec{u}$ possui um único vetor oposto.


    \questao{} Determine as coordenadas da extremidade do segmento orientado que representa o vetor $\vec{v} = (3,2)$, sabendo-se que sua origem está no ponto $P(-1,2)$.
      \begin{solucao}
        $A(2,4)$
      \end{solucao}


    \questao{} Determine as coordenadas da extremidade do segmento orientado que representa o vetor $\vec{u} = (2,-5)$, sabendo-se que sua origem está no ponto $P(-1,3)$.
      \begin{solucao}
        $A(1, -2)$.
      \end{solucao}


    \questao{} Sejam $\vec{u} = (3, -1)$ e $\vec{v} = (-1, 2)$. Determine um vetor $\vec{w}$ tal que
      \begin{enumerate}[label=({\alph*})]
        \item $4(\vec{u} - \vec{v}) + \vec{w}/3 = 2\vec{u} - \vec{w}$
        \item $3\vec{w} - (2\vec{v} - \vec{u}) = 2(4\vec{w} - 3\vec{u})$
      \end{enumerate}
      \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
          \item $\vec{w} = (-15/2, 15/2)$
          \item $\vec{w} = (23/5, -11/5)$
        \end{enumerate}
      \end{solucao}


    \questao{} Sejam $\vec{u} = (2, -4)$, $\vec{v} = (-5, 1)$ e $\vec{w} = (-12, 6)$. Encontre escalares $k_1$ e $k_2$ tais que $\vec{w} = k_1\vec{u} + k_2\vec{v}$.
      \begin{solucao}
        $k_1 = -1$ e $k_2 = 2$
      \end{solucao}


    \questao{} Sejam $\vec{u} = (a, b)$ e $\vec{v} = (x, y)$ vetores em $\real^2$. Mostre que:
      \begin{enumerate}[label=({\alph*})]
        \item $\vec{u}$ e $\vec{v}$ são perpendiculares se, e somente se, $\vec{u} = \vec{0}$ ou se existir $\lambda \in \real$ tal que $x = -\lambda b$ e $y = \lambda a$.

        \item $\vec{u}$ e $\vec{v}$ são paralelos se, e somente se, existir $\lambda \in \real$ tal que $\vec{v} = \lambda\vec{u}$ ou $\vec{u} = \lambda\vec{v}$.

        \item $\vec{u}$ e $\vec{v}$ são paralelos se, e somente se, existirem $\lambda$, $\alpha \in \real$ não simultaneamente nulos tais que $\lambda\vec{u} + \alpha\vec{v} = \vec{0}$.
      \end{enumerate}


    \questao{} Sejam $\vec{u}$ e $\vec{v} \ne \vec{0}$ vetores em $\real^2$. Se $\vec{u}$ é paralelo e ortogonal a $\vec{v}$, então $\vec{u} = \vec{0}$.


    \questao{} Os lados de um triângulo equilátero $ABC$ têm medida 2. Calcule $\inner{AB}{BC} + \inner{BC}{CA} + \inner{CA}{AB}$.
      \begin{solucao}
        $\inner{AB}{BC} + \inner{BC}{CA} + \inner{CA}{AB} = -6$
      \end{solucao}

    \questao{} Para os vetores $\vec{u}$ e $\vec{v}$ dados a seguir, calcule o ângulo $\theta$, em radianos, entre $\vec{u}$ e $\vec{v}$.
      \begin{enumerate}[label=({\alph*})]
        \item $\vec{u} = (1, 0)$ e $\vec{v} = (-2, 1)$
        \item $\vec{u} = (3, 3)$ e $\vec{v} = (2, 1)$
        \item $\vec{u} = (\sqrt{3}/2, 1/2)$ e $\vec{v} = (\sqrt{3}/2, -1/2)$
        \item $\vec{u} = (-1, 1)$ e $\vec{v} = (1, 1)$
      \end{enumerate}
      \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
          \item $\theta = \arccos(-2/\sqrt{5})$
          \item $\theta = \arccos(3/\sqrt{10})$
          \item $\theta = \pi/3$
          \item $\theta = \pi/2$
        \end{enumerate}
      \end{solucao}


    \questao{} Determine $x$ de modo que $\vec{u}$ e $\vec{v}$ sejam ortogonais.
      \begin{enumerate}[label=({\alph*})]
        \item $\vec{u} = (x, 0)$ e $\vec{v} = (1, x)$
        \item $\vec{u} = (x, x)$ e $\vec{v} = (4, x)$
        \item $\vec{u} = (x + 1, 1)$ e $\vec{v} = (x - 1, -1)$
        \item $\vec{u} = (x, -1)$ e $\vec{v} = (x, -3)$
      \end{enumerate}
      \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
          \item $x = 0$
          \item $x = 0$ ou $x = -4$
          \item $x = \pm\sqrt{2}$
          \item Não existe $x \in \real$ tal que $\vec{u}$ e $\vec{v}$ são ortogonais.
        \end{enumerate}
      \end{solucao}


    \questao{} Verdadeiro ou falso? Justifique.
      \begin{enumerate}[label=({\alph*})]
        \item $\inner{u}{u} = 0$ se, e somente se, $\vec{u} = \vec{0}$.
        \item $\langle\vec{u}, \vec{v} - \vec{w}\rangle = \inner{u}{v} - \inner{u}{w}$.
        \item Se $\inner{u}{v} = 0$, então $\vec{u} = \vec{0}$ ou $\vec{v} = \vec{0}$.
        \item Se $\vec{u} = -\vec{v}$, então $\inner{u}{v} \le 0$.
      \end{enumerate}
      \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
          \item Verdadeiro
          \item Verdadeiro
          \item Falso
          \item Verdadeiro
        \end{enumerate}
      \end{solucao}


    \questao{} Sabendo que $\vec{u} + \vec{v} + \vec{w} = \vec{0}$, $\norm{\vec{u}} = 3/2$, $\norm{\vec{v}} = 1/2$ e $\norm{\vec{w}} = 2$, calcule $\inner{u}{v} + \inner{v}{w} + \inner{w}{u}$.
      \begin{solucao}
        $\inner{u}{v} + \inner{v}{w} + \inner{w}{u} = -13/4$
      \end{solucao}


    \questao{} Sejam $\vec{e_1} = (1, 0)$ e $\vec{e_2} = (0, 1)$. Dado um vetor $\vec{u}$, mostre que $\inner{u}{e_1} = x$ e $\inner{u}{e_2} = y$. Conclua que $\vec{u} = \inner{u}{e_1}\vec{e_1} + \inner{u}{e_2}\vec{e_2}$.


    \questao{} Sejam $\vec{u} = (2, t)$ e $\vec{v} = (-t^2, 2)$. Calcule $t$ para que:
      \begin{enumerate}[label=({\alph*})]
        \item $\vec{u}$ e $\vec{v}$ sejam paralelos;
        \item $\vec{u}$ e $\vec{v}$ sejam perpendiculares.
      \end{enumerate}
      \begin{solucao}
      \begin{enumerate}[label=({\alph*})]
        \item $t = -\dfrac{2}{\sqrt[3]{2}}$
        \item $t = 0$ ou $t = 1$
      \end{enumerate}
    \end{solucao}

    \questao{} Na Figura \ref{circunferencia}, a circunferência de centro $O$ tem raio $r$. Calcule $\inner{BA}{BC}$ em fun\c{c}ão de $r$ e dos ângulos $\alpha$ e $\beta$, indicados.
      \begin{figure}
        \centering
        \caption{Circunferência}
        \includegraphics{exercicio_circunferencia.pdf}
        \label{circunferencia}
      \end{figure}
      \begin{solucao}
        $\inner{BA}{BC} = r^2[1 + \cos\alpha - \cos\beta - \cos(\alpha + \beta)]$
      \end{solucao}


    \questao{} Prove que
      \begin{enumerate}[label=({\alph*})]
        \item $4\inner{u}{v} = \norm{\vec{u} + \vec{v}}^2 - \norm{\vec{u} - \vec{v}}^2$.
        \item $\inner{u}{v} = 0$ se, e somente se, $\norm{\vec{u} + \vec{v}} = \norm{\vec{u} - \vec{v}}$.
        \item $\langle\vec{u} + \vec{v}, \vec{u} - \vec{v}\rangle = \norm{\vec{u}}^2 - \norm{\vec{v}}^2$.
        \item $\norm{\vec{v}}\vec{u} + \norm{\vec{u}}\vec{v}$ é ortogonal a $\norm{\vec{v}}\vec{u} - \norm{\vec{u}}\vec{v}$.
      \end{enumerate}

    \questao{} Prove que se $\vec{u} \ne \vec{0}$, então $\proj_{\vec{u}}(\alpha\vec{v} + \beta\vec{w}) = \alpha\proj_{\vec{u}}\vec{v} + \beta\proj_{\vec{u}}\vec{w}$.



    \questao{} Dado o triângulo cujos vértices são $A(1, 1)$, $B(4, 0)$ e $C(3, 4)$ determine:
    \begin{enumerate}[label=({\alph*})]
        \item os ângulos $A$, $B$ e $C$.
        \item o comprimento das proje\c{c}\~oes dos lados $AC$ e $BC$ sobre o lado $AB$.
        \item o pé da altura relativo ao vértice $C$.
      \end{enumerate}
      \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
          \item ângulo entre $\vec{AC}$ e $\vec{AB}$: $\arccos\left(\dfrac{3}{\sqrt{130}}\right)$; ângulo entre $\vec{BA}$ e $\vec{BC}$: $\arccos\left(\dfrac{7}{\sqrt{170}}\right)$; ângulo entre $\vec{CA}$ e $\vec{CB}$: $\arccos\left(\dfrac{10}{\sqrt{221}}\right)$
          \item $\norm{\proj_{\vec{AB}}\vec{AC}} = \dfrac{3}{10}\sqrt{10}$; $\norm{\proj_{\vec{AB}}\vec{BC}} = \dfrac{7}{10}\sqrt{10}$
          \item $P(19/10, 7/10)$
        \end{enumerate}
      \end{solucao}


    \questao{} Determine a altura relativa ao lado $AD$ do paralelogramo de vértices $A(1,0)$, $B(2,2)$, $C(5,3)$ e $D(4,1)$.
      \begin{solucao}
        $\dfrac{\sqrt{10}}{2}$
      \end{solucao}


    \questao{} Sejam $\proj_{\vec{v}}\vec{u} = (2, 1)$, $\vec{u} = (4,2)$, $\norm{\vec{v}} = 6$. Determine $\vec{v}$.

    \questao{} Calcule as seguintes áreas:
      \begin{enumerate}[label=({\alph*})]
        \item Do triângulo de vértices $(1, 2)$, $(2, -1)$ e $(6, 8)$.
        \item Do equilátero de vértices $(1, 1)$, $(3, 1)$, $(2, 5)$ e $(6, 6)$.
        \item Do paralelogramo cujas lados são paralelos a $(1, 3)$ e $(2, -1)$.
        \item Do pol{\'\i}gono cujos vértices são $(0, 0)$, $(1, 1)$, $(2, 0)$, $(2, 2)$, $(1, 3)$ e $(0, 2)$.
        \item Do paralelogramo que admite vértices $(1, 1)$, $(2, -1)$ e $(1, 3)$.
      \end{enumerate}
      \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
          \item $A = 21/2$
          \item $A = 25/2$
          \item $A = 7$
          \item $A = 4$
          \item $A = 2$
        \end{enumerate}
      \end{solucao}
