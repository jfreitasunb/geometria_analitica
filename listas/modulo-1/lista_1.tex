    \begin{enumerate}[label=({\arabic*})]
        \item Esfera: $x^2 + y^2 + z^2 = r^2$
        \item Elipsoide: $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} + \dfrac{z^2}{c^2} = 1$
        \item Cilindro elíptico: $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$
        \item Paraboloide elíptico: $z = \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2}$, $a \ne b$
        \item Paraboloide de rotação: $z = \dfrac{x^2}{a^2} + \dfrac{y^2}{a^2}$
        \item Paraboloide hiperbólico: $z = \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2}$
        \item Hiperboloide de uma folha: $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} - \dfrac{z^2}{c^2} = 1$
        \item Hiperboloide de duas folhas: $\dfrac{z^2}{c^2} - \dfrac{y^2}{b^2} - \dfrac{x^2}{a^2} = 1$
        \item Cilindro hiperbólico: $\dfrac{y^2}{b^2} - \dfrac{x^2}{a^2} = 1$
        \item Cilindro parabólico: $y = ax^2$
        \item Cone elíptico: $z^2 = \dfrac{x^2}{a^2} + \dfrac{y^2}{b^2}$, $a \ne b$
        \item Cone circular: $z^2 = \dfrac{x^2}{a^2} + \dfrac{y^2}{a^2}$
    \end{enumerate}
