    Uma \textbf{esfera} de centro $(x_0,y_0,z_0)$ e raio $r > 0$ é o conjunto de pontos $(x,y,z)$ do espaço que satisfaz a equação
    \[
    (x - x_0)^2 + (y - y_0)^2 + (z - z_0)^2 = r^2.
    \]

    \begin{exercicio}
        Determine o centro e o raio das seguintes esferas:
        \begin{enumerate}[label=({\alph*})]
            \item $x^2 + y^2 + z^2 - 2x - 4y - 2z = 10$
            \item $2x^2 + 2y^2 + 2z^2 - 2x + 6y = 6$
            \item $x^2 + y^2 + z^2 = 3$
        \end{enumerate}
        \begin{solucao}
            Complete quadrados.
            \begin{enumerate}[label=({\alph*})]
                \item $C(1,2,1)$, $r = 4$
                \item $C(1/2,-3/2,0)$, $r = \sqrt{22}{4}$
                \item $C(0,0,0)$, $r = \sqrt{3}$
            \end{enumerate}
        \end{solucao}
    \end{exercicio}

    \begin{exercicio}
        Determine uma equação da esfera que tem por diâmetro o segmento de extremos $A(8,0,3)$ e $B(-6,2,5)$.
        \begin{solucao}
            $(x - 1)^2 + (y - 1)^2 + (z - 4)^2 = 51$
        \end{solucao}
    \end{exercicio}

    \begin{exercicio}
        Verifique que quaisquer que sejam os valores de $\theta$ e $\alpha$, $0 \le \theta \le 2\pi$ e $0 \le \alpha \le \pi$ o ponto
        \[
        (r\sin\alpha\cos\theta, r\sin\alpha\sin\theta, r\cos\alpha)
        \]
        pertence à esfera de raio $r$ e centro na origem.
    \end{exercicio}

    \begin{exercicio}
        Identifique as quádricas abaixo:
        \begin{enumerate}[label=({\alph*})]
            \item $2x^2 + 30y^2 + 23z^2 + 72xy + 150 = 0$
            \item $144x^2 + 100y^2 + 81z^2 - 216 xz - 540x - 720z = 0$
            \item $2xy + z = 0$
            \item $2xy + 2xz + 2yz - 6x - 6y - 4z - 9 = 0$
            \item $7x^2 + 7y^2 + 10z^2 - 2xy - 4xz + 4yz - 12x + 12y + 60z - 24 = 0$
            \item $5x^2 + 6y^2 + 7z^2 - 4xy - 4yz - 18x - 12y + 6z - 18 = 0$
            %\item $x^2 - y^2 + 4xy + 4yz + 8x - 2y + 11z + 12 = 0$
            \item $x^2 + y^2 + z^2 - 2xy + 2xz - 2yz - 4x + 4y - 4z + 1 = 0$
            \item $x^2 + 2y^2 + 2z^2 + 2xz + 2xy + 6x - 2y + 2z + 14 = 0$
            \item $3x^2 + 2y^2 + 2z^2 - 6xy - 6xz - 4yz + 10x + 2y - 6z - 8 = 0$
        \end{enumerate}
        \begin{solucao}
            \begin{enumerate}[label=({\alph*})]
                \item Hiperbolóide de duas folhas: $x_1^2/6 - y_1^2/5 - z_1^2/3 = 1$
                \item Parabolóide: $y_1^2/9 + z_1^2/4 - x_1 = 0$
                \item Parabolóide hiperbólico: $y_1^2 - z_1^2 + x_1 = 0$
                \item Hiperbolóide de duas folhas: $-x_2^2 - y_2^2 + 2z_2^2 + 1 = 0$
                \item Elipsóide: $x_2^2/19 + y_2^2/19 + 2z_2^2/19 = 1$
                \item Elipsóide.
                %\item Hiperbolóide de duas folhas.
                \item Um par de planos paralelos.
                \item Parabolóide.
                \item Hiperbolóide de uma folha.
            \end{enumerate}
        \end{solucao}
    \end{exercicio}
