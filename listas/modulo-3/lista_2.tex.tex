\begin{exercicio}
    Identifique as qu\'adricas abaixo:
    \begin{enumerate}[label=({\alph*})]
        \item $2x^2 + 30y^2 + 23z^2 + 72xy + 150 = 0$
        \item $144x^2 + 100y^2 + 81z^2 - 216 xz - 540x - 720z = 0$
        \item $2xy + z = 0$
        \item $2xy + 2xz + 2yz - 6x - 6y - 4z - 9 = 0$
        \item $7x^2 + 7y^2 + 10z^2 - 2xy - 4xz + 4yz - 12x + 12y + 60z - 24 = 0$
        \item $5x^2 + 6y^2 + 7z^2 - 4xy - 4yz - 18x - 12y + 6z - 18 = 0$
        %\item $x^2 - y^2 + 4xy + 4yz + 8x - 2y + 11z + 12 = 0$
        \item $x^2 + y^2 + z^2 - 2xy + 2xz - 2yz - 4x + 4y - 4z + 1 = 0$
        \item $x^2 + 2y^2 + 2z^2 + 2xz + 2xy + 6x - 2y + 2z + 14 = 0$
        \item $3x^2 + 2y^2 + 2z^2 - 6xy - 6xz - 4yz + 10x + 2y - 6z - 8 = 0$
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
            \item Hiperbol\'oide de duas folhas: $x_1^2/6 - y_1^2/5 - z_1^2/3 = 1$
            \item Parabol\'oide: $y_1^2/9 + z_1^2/4 - x_1 = 0$
            \item Parabol\'oide hiperb\'olico: $y_1^2 - z_1^2 + x_1 = 0$
            \item Hiperbol\'oide de duas folhas: $-x_2^2 - y_2^2 + 2z_2^2 + 1 = 0$
            \item Elips\'oide: $x_2^2/19 + y_2^2/19 + 2z_2^2/19 = 1$
            \item Elips\'oide.
            %\item Hiperbol\'oide de duas folhas.
            \item Um par de planos paralelos.
            \item Parabol\'oide.
            \item Hiperbol\'oide de uma folha.
        \end{enumerate}
    \end{solucao}
