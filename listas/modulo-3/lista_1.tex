\begin{exercicio}
    Escreva uma equação do plano:
    \begin{enumerate}[label=({\alph*})]
        \item que contém o ponto $(1,1,1)$ e é perpendicular ao vetor $(2,-1,8)$;
        \item que contém os pontos $A(2,-1,3)$, $B(0,2,1)$ e $C(1,3,2)$;
        \item que contém os pontos $A(0,0,2)$, $B(1,2,2)$ e $C(1,0,2)$;
        \item contendo o ponto $(2,1,3)$ e a interseção do plano $2x - y - z = 2$ com o plano $xy$;
        \item paralelo ao plano $yz$ e que contém o ponto $(3,4,1)$;
        \item cujas interseções com os eixos coordenados são os pontos $(3,0,0)$, $(0,-2,0)$ e $(0,0,-3)$.
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
            \item $2x - y + 8z = 9$
            \item $x - z + 1 = 0$
            \item $z = 2$
            \item $6x - 3y - z = 6$
            \item $x = 3$
            \item $2x - 3y - 2z = 6$
        \end{enumerate}
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Escreva as equações paramétricas da reta:
    \begin{enumerate}[label=({\alph*})]
        \item contendo os pontos $A(2,1,3)$ e $B(1,3,7)$;
        \item contendo os pontos $A(1,1,0)$ e $B(2,2,0)$;
        \item que contém o ponto $A(2,1,0)$ e é perpendicular ao plano $2x - y + z = 0$;
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
            \item $\begin{cases}
                          x = 2 - t  \\
                          y = 1 + 2t \\
                          z = 3 + 4t
                      \end{cases}$
            \item $\begin{cases}
                          x = 1 + t \\
                          y = 1 + t \\
                          z = 0
                      \end{cases}$
            \item $\begin{cases}
                          x = 2 + 2t \\
                          y = 1 - t  \\
                          z = t
                      \end{cases}$
        \end{enumerate}
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Determine a interseção entre os seguintes planos:
    \begin{enumerate}[label=({\alph*})]
        \item $\pi_1 : 2x + y - z = 0$ e $\pi_2 : x + y + z = 1$
        \item $\pi_1 : x + 2y = 1$ e $\pi_2 : z = 2$
        \item $\pi_1 : x + 2y - 3z = 0$ e $\pi_2 : x - 4y + 2z + 1 = 0$
        \item $\pi_1 : 2x - y + 4z + 3 = 0$ e $\pi_2 : 4x - 2y + 8z = 0$
        \item $\pi_1 : x - y = 0$ e $\pi_2 : x + z = 0$
        \item $\pi_1 : 2x + y + z - 1 = 0$, $\pi_2 : x + 3y + z - 2 = 0$ e $\pi_3 : x + y + 4z - 3 = 0$
        \item $\pi_1 : x - 2y + z = 0$, $\pi_2 : 2x - 4y + 2z = 1$ e $\pi_3 : x + y = 0$
        \item $\pi_1 : x - 2y + 3z = 2$, $\pi_2 : 3x + y - 2z = 1$ e $\pi_3 : 5x -3y + 4z = 4$
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
            \item $\pi_1 \cap \pi_2 : \begin{cases}
                          x = -1 + 2t \\
                          y = 2 - 3t  \\
                          z = t
                      \end{cases}$
            \item $\pi_1 \cap \pi_2 : \begin{cases}
                          x = 1 + 2t \\
                          y = - t    \\
                          z = 2
                      \end{cases}$
            \item $\pi_1 \cap \pi_2 : \dfrac{x - 4}{-8} = \dfrac{y}{-5} = \dfrac{z}{-6}$
            \item $\pi_1 \cap \pi_2 = \emptyset$
            \item $\pi_1 \cap \pi_2 : \dfrac{x - 1}{-1} = \dfrac{y}{-1} = z$
            \item $\pi_1 \cap \pi_2 \cap \pi_3 = \{(-1/17, 8/17, 11/17)\}$
            \item $\pi_1 \cap \pi_2 \cap \pi_3 = \emptyset$; $\pi_1$ e $\pi_2$ são paralelos distintos; $\pi_3 \cap \pi_1 : x = t,\ y = -t,\ z = -3t$ e $\pi_3 \cap \pi_2 : x = t\ y = -t,\ z = 1/2 - 3t$.
            \item $\pi_1 \cap \pi_2 \cap \pi_3 = \emptyset$, mas $\pi_1 \cap \pi_2 : x = -4/7 + 43t/7,\ y = -5/7 + 11t/7,\ z = t $, $\pi_1 \cap \pi_3 : x = 2/7 + t/7,\ y = -6/7 + 11t/7,\ z = t$ e $\pi_2 \cap \pi_3 : x = 1/6 - 4t/21,\ y = 1/2 + 11t/7,\ z = t$
        \end{enumerate}
    \end{solucao}
\end{exercicio}

\begin{exercicio}

    \[
        r: \begin{cases}
            x = 1 + t \\
            y = -2    \\
            z = 4 + 2t
        \end{cases}
    \]
    com cada um dos seguintes planos:
    \begin{enumerate}[label=({\alph*})]

        \item $2x + z = 5$;

            \item $(3/4,-2,7/2)$
            \item $(2,-2,6)$
        \end{enumerate}
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Determine os valores reais de $\alpha$ e $\beta$ para que as retas
    \[
        r: \begin{cases}
            x = 1 + \alpha t \\
            y = 2 + \beta t  \\
            z = -1 + 2t
        \end{cases},\quad
        s: \begin{cases}
            x = 2 + s       \\
            y = 1 + \beta s \\
            z = -1 + 2s
        \end{cases}
    \]
    sejam:
    \begin{enumerate}[label=({\alph*})]
        \item paralelas;
        \item concorrentes;
        \item reversas.
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
            \item $\alpha = 1$ e $\beta$ qualquer;
            \item impossível;
            \item $\alpha \ne 1$ e $\beta$ qualquer.
        \end{enumerate}
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Estude a posição relativa das retas $r$ e $s$.
    \begin{enumerate}[label=({\alph*})]
        \item $r: \begin{cases}
                      x = \lambda   \\
                      y = - \lambda \\
                      z = 1 + 4\lambda
                  \end{cases} \quad s: \dfrac{x - 1}{3} = \dfrac{y - 5}{3} = \dfrac{2 + z}{5}$;
        \item $r: \begin{cases}
                      x = 2 - 2\lambda \\
                      y = 4 + \lambda  \\
                      z = -3\lambda
                  \end{cases}\quad s: \begin{cases}
                      x = 1 + t \\
                      y = -2t   \\
                      z = 2t
                  \end{cases}$
        \item $r : X = (1,-1,1) + \lambda(-2,1,-1)$, $s : \begin{cases}
                      y + z = 3 \\
                      x + y - z = 6
                  \end{cases}$
        \item $r : \dfrac{x + 1}{2} = \dfrac{y}{3} = \dfrac{z + 1}{2}$, $s : X = (0,0,0) + \lambda(1,2,0)$
        \item $r : \dfrac{x + 3}{2} = \dfrac{y - 1}{4} = z$. $s : \begin{cases}
                      2x - y + 7 = 0 \\
                      x + y - 6z = -2
                  \end{cases}$
        \item $r : x + 3 = \dfrac{2y - 4}{4} = \dfrac{z - 1}{3}$, $s : X = (0,2,2) + \lambda(1,1,-1)$
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
            \item Concorrentes no ponto $(-2,2,-7)$.
            \item Concorrentes no ponto $(-2,6,-6)$.
            \item Paralelas distintas.
            \item Reversas.
            \item Coincidentes.
            \item Reversas.
        \end{enumerate}
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Seja $r$ a reta $r : (x,y,z) = (1,0,1) + t(1,-1,0)$ e $\pi$ o plano $\pi : x + y + z = 0$.
    \begin{enumerate}[label=({\alph*})]
        \item Verifique que $r \varparallel \pi$.
        \item Calcule a distância de $r$ a $\pi$.
        \item Existem retas contidas no plano $\pi$, que são reversas à reta $r$ e distam 2 desta?
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}
            \item[(b)] $d(r,\pi) = 2/\sqrt{3}$
            \item[(c)] Não
        \end{enumerate}
    \end{solucao}
\end{exercicio}
