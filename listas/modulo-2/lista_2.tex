Sejam $\vec{i} = (1, 0, 0)$, $\vec{j} = (0, 1, 0)$ e $\vec{k} = (0, 0, 1)$ vetores em $\real^3$.

\begin{exercicio}
    Sejam $\vec{v} = \vec{i} + 2\vec{j} - 3\vec{k}$ e $ \vec{w} = 2\vec{i} + \vec{j} - 2\vec{k}$. Determine vetores unit\'arios paralelos aos vetores:
    \begin{enumerate}[label=({\alph*})]
        \item $\vec{v} + \vec{w}$
        \item $\vec{v} - \vec{w}$
        \item $2\vec{v} - 3\vec{w}$
    \end{enumerate}
\end{exercicio}

\begin{exercicio}
    Determine o valor de $x$ para o qual os vetores $\vec{v} = x\vec{i} + 3\vec{j} + 4\vec{k}$ e $\vec{w} = 3\vec{i} + \vec{j} + 2\vec{k}$ s\~ao perpendiculares.
    \begin{solucao}
        $x = -11/3$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Demonstre que n\~ao existe $x$ tal que os vetores $\vec{v} = x\vec{i} + 2\vec{j} + 4\vec{k}$ e $\vec{w} = x\vec{i} - 2\vec{j} + 3\vec{k}$ s\~ao perpendiculares.
\end{exercicio}

\begin{exercicio}
    Dados os vetores
    \[
        \vec{u} = (2,-3,1),\quad \vec{v} = (2,2,0) \text{ e } \vec{w} = (1,-3,4),
    \]
    calcule:
    \begin{enumerate}[label=({\alph*})]
        \item $\vec{u}\times\vec{v}$
        \item $(\vec{u}\times\vec{v})\times\vec{w}$
        \item $(\vec{u}\times\vec{v})\times(\vec{u}\times\vec{w})$
        \item $(\vec{u} + \vec{v})\times(\vec{u} + \vec{w})$
    \end{enumerate}
    \begin{solucao}
        \begin{enumerate}[label=({\alph*})]
            \item $\vec{u}\times\vec{v} = (-2,2,10)$
            \item $(\vec{u}\times\vec{v})\times\vec{w} = (38,18,4)$
            \item $(\vec{u}\times\vec{v})\times(\vec{u}\times\vec{w}) = (64,-96,32)$
            \item $(\vec{u} + \vec{v})\times(\vec{u} + \vec{w}) = (1,-17,-21)$
        \end{enumerate}
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Calcule a \'area do paralelogramo definido pelos vetores $\vec{u} = (2,3,1)$ e $\vec{v} = (1,5,3)$.
    \begin{solucao}
        $3\sqrt{10}$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Mostre que qualquer que seja o valor de $a$, o m\'odulo do vetor $\vec{w} = (1 - a, 1, a - 2)$ \'e igual \`a \'area do paralelogramo definido pelos vetores $\vec{u} = (1,1,1)$ e $\vec{v} = (2,a,1)$.
\end{exercicio}

\begin{exercicio}
    Calcule a \'area do paralelogramo $ABCD$, sendo $\vec{AB} = (1,1,-1)$ e $\vec{AD} = (2,1,4)$.
    \begin{solucao}
        $\sqrt{62}$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Sejam $\vec{i} = (1,0,0)$, $\vec{j} = (0,1,0)$ e $\vec{k} = (0,0,1)$. Mostre que
    \[
        \vec{i}\times\vec{j} = \vec{k},\quad \vec{j}\times\vec{k} = \vec{i},\quad \vec{k}\times\vec{i} = \vec{j}.
    \]
\end{exercicio}

\begin{exercicio}
    Sejam $\vec{u}$ e $\vec{v}$ vetores unit\'arios e perpendiculares entre si. Mostre que $\norm{\vec{u}\times\vec{v}} = 1$.
\end{exercicio}

\begin{exercicio}
    Encontre $\vec{v}$ tal que $\vec{v} \times (\vec{i} + \vec{j}) = 2(\vec{i} + \vec{j} - \vec{k})$ e $\norm{\vec{v}} = \sqrt{6}$.
    \begin{solucao}
        $\vec{v} = (-1, 2, 1)$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Calcule a \'area do tri\^angulo de v\'ertices $A(1,2,1)$, $B(3,0,4)$ e $C(5,1,3)$.
    \begin{solucao}
        \'Area = $\sqrt{101}/2$.
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Encontre um vetor $\vec{u}$ tal que $\vec{u}\times(\vec{i} + \vec{k}) = 2(\vec{i} + \vec{j} - \vec{k})$ e $\norm{\vec{u}} = \sqrt{6}$.
    \begin{solucao}
        $\vec{u} = (-1,2,1)$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Sabendo que $\norm{\vec{u}} = 1$, $\norm{\vec{v}} = 7$ e que o \^angulo entre $\vec{u}$ e $\vec{v}$ \'e $\pi/6$, calcule $\norm{\vec{u}\times\vec{v}}$ e $\norm{4\vec{u}\times9\vec{v}}$.
    \begin{solucao}
        $\norm{\vec{u}\times\vec{v}} = 7/2$, $\norm{4\vec{u}\times9\vec{v}} = 126$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    O \^angulo entre os vetores $\vec{a}$ e $\vec{b}$ \'e $\pi/3$, e suas normas s\~ao, respectivamente, 1 e 2. Sendo $\vec{u} = \vec{a} + \vec{b}$ e $\vec{v} = \vec{a} - \vec{b}$, calcule a norma de $\norm{\vec{u}\times\vec{v}}$.
    \begin{solucao}
        $\norm{\vec{u}\times\vec{v}} = 2\sqrt{3}$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Sejam $\vec{v}$ um vetor n\~ao nulo no espa\c{c}o e $\alpha$, $\beta$ e $\gamma$ os \^angulos que $\vec{v}$ formam com os vetores $\vec{i}$, $\vec{j}$ e $\vec{k}$, respectivamente. Demonstre que
    \[
        \cos^2\alpha + \cos^2\beta + \cos^2\gamma = 1.
    \]
\end{exercicio}

\begin{exercicio}
    Sejam $\vec{U_1}$, $\vec{U_2}$ e $\vec{U_3}$ tr\^es vetores unit\'arios e dois a dois ortogonais. Se $A = [\vec{U_1}\ \vec{U_2}\ \vec{U_3}]$ \'e uma matriz $3 \times 3$ cujas colunas s\~ao os vetores $\vec{U_1}$, $\vec{U_2}$ e $\vec{U_3}$, ent\~ao $A$ \'e invert{\'\i}vel e $A^{-1} = A^t$. \textit{(Sugest\~ao: mostre que $A^tA = I_3$.)}
\end{exercicio}

\begin{exercicio}
    Sejam $\vec{u} = (x_1, y_1, z_1)$, $\vec{v} = (x_2, y_2, z_2)$ e $\vec{w} = (x_3, y_3, z_3)$ vetores em $\real^3$. Prove a f\'ormula seguinte para o \textbf{produto vetorial duplo}
    \[
        \vec{u}\times (\vec{v}\times \vec{w}) = \inner{u}{w}\vec{v} - \inner{u}{v}\vec{w},
    \]
    seguindo os seguintes passos:
    \begin{enumerate}[label=({\alph*})]
        \item Prove que
              \begin{align*}
                  \vec{u} \times (\vec{i}\times\vec{j}) & = \inner{u}{j}\vec{i} - \inner{u}{i}\vec{j} \\
                  \vec{u} \times (\vec{j}\times\vec{k}) & = \inner{u}{k}\vec{j} - \inner{u}{j}\vec{k} \\
                  \vec{u} \times (\vec{k}\times\vec{i}) & = \inner{u}{i}\vec{k} - \inner{u}{k}\vec{i}
              \end{align*}

        \item Prove usando o item anterior e as propriedades do produto vetorial que
              \begin{align*}
                  \vec{u} \times (\vec{v}\times\vec{i}) & = \inner{u}{i}\vec{v} - \inner{u}{v}\vec{i} \\
                  \vec{u} \times (\vec{v}\times\vec{j}) & = \inner{u}{j}\vec{v} - \inner{u}{v}\vec{j} \\
                  \vec{u} \times (\vec{v}\times\vec{k}) & = \inner{u}{k}\vec{v} - \inner{u}{v}\vec{k}
              \end{align*}

        \item Prove agora o caso geral usando o item anterior e as propriedades do produto vetorial.
    \end{enumerate}
\end{exercicio}

\begin{exercicio}
    Prove usando o exerc{\'\i}cio anterior que
    \[
        [\vec{u}\times (\vec{v}\times \vec{w})] + [\vec{v}\times (\vec{w}\times \vec{u})] + [\vec{w}\times (\vec{u}\times \vec{v})] = \vec{0}.
    \]
\end{exercicio}

\begin{exercicio}
    Mostre que n\~ao existe $x \in \real$ tal que os vetores $\vec{u} = (x,2,4)$ e $\vec{v} = (x,-2,3)$ s\~ao perpendiculares.
\end{exercicio}

\begin{exercicio}
    Sejam $\vec{u} = (-1,1,-3)$, $\vec{v} = (1,1,0)$ e $\vec{w} = (2,1,1)$. Calcule $(\vec{u}\times\vec{v})\cdot\vec{w}$.
    \begin{solucao}
        $(\vec{u}\times\vec{v})\cdot\vec{w} = 1$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    O \^angulo entre os vetores unit\'arios $\vec{u}$ e $\vec{v}$ \'e $\pi/6$, e o vetor $\vec{w}$, de norma 4, \'e ortogonal a ambos. Calcule $\langle(\vec{u}\times\vec{v}), \vec{w}\rangle$.
    \begin{solucao}
        $(\vec{u}\times\vec{v})\cdot\vec{w} = 2$
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Sejam $A$, $B$ e $C$ pontos n\~ao-colineares. Exprima a dist\^ancia de um ponto $D$ ao plano $ABC$ em fun\c{c}\~ao de $\vec{AB}$, $\vec{AC}$, $\vec{AD}$.
    \begin{solucao}
        $|(\vec{AB}\times\vec{AC})\cdot\vec{AD}|/\norm{\vec{AB}\times\vec{AC}}$. [Sugest\~ao: utilize o volume do paralelep{\'\i}pedo definido por $A$, $B$, $C$, $D$.]
    \end{solucao}
\end{exercicio}

\begin{exercicio}
    Calcule o volume do paralelep{\'\i}pedo que tem um dos v\'ertices no ponto $A(2,1,6)$ e o tr\^es v\'ertices adjacentes nos pontos $B(4,1,3)$, $C(1,3,2)$ e $D(1,2,1)$.
    \begin{solucao}
        Volume = 15.
    \end{solucao}
\end{exercicio}
