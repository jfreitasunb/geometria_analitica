\begin{exercicio}
	Escreva a equa\c{c}\~ao da circunfer\^encia de centro $P_0$ e raio $r$, onde:
	\begin{enumerate}[label=({\alph*})]
		\item $P_0(1,0)$ e $r = 2$
		\item $P_0(0,0)$ e $r$ arbitr\'ario
		\item $P_0(2,-1)$ e $r = \sqrt{2}$
	\end{enumerate}
	\begin{solucao}
		\begin{enumerate}[label=({\alph*})]
			\item $x^2 + y^2 -2x - 3 = 0$
			\item $x^2 + y^2 - r^2 = 0$
			\item $x^2 + y^2 - 4x + 2y + 3 = 0$
		\end{enumerate}
	\end{solucao}
\end{exercicio}

\begin{exercicio}
	Nos casos em que a equa\c{c}\~ao dada descreve uma elipse de focos em algum dos eixos coordenados, especifique-o e calcule: a dist\^ancia focal, a medida do eixo maior e a medida do eixo menor. Fa\c{c}a um esbo\c{c}o da elipse.
	\begin{enumerate}[label=({\alph*})]
		\item $4x^2 + 169y^2 = 676$
		\item $x^2 + 2y^2/3 = 8$
		\item $x^2/4 + y^2/2 = 0$
		\item $4x^2 + 9y^2 + 1 = 0$
		\item $16x^2 - 4 + 4y^2 = 0$
		\item $x^2 + m^2y^2 = m$
	\end{enumerate}
	\begin{solucao}
		\begin{enumerate}[label=({\alph*})]
			\item Eixo $x$, 26, 4, $2\sqrt{165}$.
			\item Eixo $y$, $4\sqrt{3}$, $4\sqrt{2}$,4.
			\item N\~ao se trata de uma elipse: s\'o a origem satisfaz a equa\c{c}\~ao.
			\item N\~ao se trata de uma elipse: \'e o conjunto vazio.
			\item Eixo $y$, 2, 1, $\sqrt{3}$.
			\item Se $m > 1$, \'e uma elipse com focos no eixo $x$, $2\sqrt{m}$, $2/\sqrt{m}$, $2\sqrt{(m^2 - 1)/m}$.
			      Se $0 < m < 1$, \'e uma elipse com focos no eixo $y$, $2/\sqrt{m}$, $2\sqrt{m}$, $2\sqrt{(1 - m^2)/m}$.
			      Se $m = 0$ \'e o eixo $y$. Se $m = 0$, \'e o conjunto vazio. E se $m = 1$, \'e uma circunfer\^encia.
		\end{enumerate}
	\end{solucao}
\end{exercicio}

\begin{exercicio}
	Deduza a equa\c{c}\~ao da reta perpendicular \`a reta que \'e tangente \`a elipse
	\[
		\elipse{a^2}{b^2}
	\]
	no ponto $(x_0,y_0)$. Esta reta \'e chamada \textbf{normal \`a elipse} em $(x_0,y_0)$.
	\begin{solucao}
		$\dfrac{y_0x}{b^2} - \dfrac{x_0y}{a^2} = x_0y_0\left(\dfrac{a^2 - b^2}{a^2b^2}\right)$
	\end{solucao}
\end{exercicio}

\begin{exercicio}
	Mostre que, qualquer que seja o valor de $t$, o ponto
	\[
		(a\cos t, b\sin t)
	\]
	pertence \`a elipse
	\[
		\elipse{a^2}{b^2}
	\]
	Estas equa\c{c}\~oes s\~ao chamadas de \textbf{equa\c{c}\~oes param\'etricas} da elipse. Quando $t$ varia de 0 a $2\pi$, o ponto $(a\cos t, b\sin t)$ percorre a elipse, a partir do v\'ertice $A_2(a,0)$ uma \'unica vez.
\end{exercicio}
