		A circunferência é o conjunto dos pontos do plano que estão a uma mesma distância (denominada \textbf{raio}) \pause de um dado ponto do plano
		(chamado \textbf{centro}).
	\end{definicao}

\begin{frame}
	\begin{definicao}
		As equações
		\[
			\begin{cases}
				x = x_0 + r\cos\beta \\
				y = y_0 + r\sin\beta,
			\end{cases}
			\beta \in [0,2\pi]
		\]\pause
		são chamadas \textbf{equações paramétricas} da circunferência \pause de centro $C(x_0, y_0)$ \pause e raio $r$.
	\end{definicao}
\end{frame}

\begin{frame}
	\begin{definicao}
		Chama-se \textbf{cônica} \pause o lugar geométrico(conjunto) \pause dos pontos $P(x,y)$ \pause que satisfazem uma equação do segundo grau $g(x,y) = 0$,
		\pause onde\index{Cônicas}
		\begin{equation}\label{equacaoconica}
			g(x,y) = ax^2 + bxy + cy^2 + dx + ey + f.\pause
		\end{equation}
	\end{definicao}
\end{frame}

\begin{frame}
	Na equação \eqref{equacaoconica} pelo menos um dos números $a$, $b$ ou $c$ \pause é diferente de zero. \pause Os termos $ax^2$ e $cy^2$ são os
	\textbf{termos quadráticos}, \pause o termo $bxy$ é o \textbf{termo quadrático misto}, \pause $dx$ e $ey$ são os \textbf{termos lineares} \pause e $f$ é o
	\textbf{termo independente}.
\end{frame}

\begin{frame}
	A equação \eqref{equacaoconica} pode representar:\pause
	\begin{enumerate}[label=({\arabic*})]
		\item um conjunto vazio: $x^2 + y^2 = -1$;\pause
		\item um conjunto formado por um ponto: $x^2 + y^2 = 0$;\pause
		\item a reunião de duas retas (paralelas ou concorrentes): $x^2 - y^2 - 2x - 6y -8 =0$;\pause
		\item uma circunferência: $x^2 + y^2 - 13x + y - 10 = 0$;\pause
		\item uma elipse: $4x^2 + 9y^2 - 8x - 36y + 4 = 0$;\pause
		\item uma hipérbole: $3x^2 + 3y^2 - 10xy + 12\sqrt{2}x - 4\sqrt{2}y + 32 = 0$;\pause
		\item uma parábola: $x^2 - 2xy + y^2 -2x - 2y + 1 = 0$.\pause
	\end{enumerate}
\end{frame}

\begin{frame}
	Nos casos em que a equação \eqref{equacaoconica} representar \pause um conjunto vazio, \pause um ponto, \pause retas ou circunferências \pause diremos que a
	cônica definida é uma \textbf{cônica degenerada}. \pause Aqui vamos estudar as cônicas chamadas de elipse, \pause hipérbole \pause e parábola.
\end{frame}

\begin{frame}
	\begin{definicao}
		Sejam $F_1$ e $F_2$ pontos distintos do plano, \pause $2c$ a distância entre $F_1$ e $F_2$ \pause e $a$ um número real \pause tal que $a > c$. \pause O
		lugar geométrico $\mathcal{E}$ \pause dos pontos $P$ tais que\pause
		\begin{equation}\label{definicaoelipse}
			d(P,F_1) \pause + d(P,F_2) \pause = 2a\pause
		\end{equation}
		chama-se \textbf{elipse}. \pause Cada um dos pontos $F_1$ e $F_2$ é chamado \textbf{foco} da elipse. \pause O segmento $F_1F_2$ é chamado
		\textbf{segmento focal}, \pause seu ponto médio é o \textbf{centro} da elipse \pause e $2c$ é chamado \textbf{distância focal}. \pause A reta passando por $F_1$ e $F_2$ chama-se \textbf{reta focal}.
	\end{definicao}
\end{frame}


\begin{frame}
	\begin{proposicao}
		Um ponto $P(x,y)$ é um ponto da elipse de equação reduzida
		\begin{align}\label{equacaoelipse}
			\elipse{a^2}{b^2}
		\end{align}\pause
		se, e somente se, \pause as distâncias de $P$ aos focos $F_1$ e $F_2$ são\pause
		\begin{align*}
			d^2(P,F_1) & = \dfrac{c}{a}\alpha + a  \\
			d^2(P,F_2) & = a - \dfrac{c}{a}\alpha.
		\end{align*}
	\end{proposicao}
\end{frame}

\begin{frame}
	\begin{observacao}
		\begin{enumerate}[label=({\arabic*})]
			\item A equação \eqref{equacaoelipse} só é válida para elipses com focos no eixo $x$ e centro na origem. \pause Mudanças nos focos ou no centro,
			      \pause resultarão em uma equação diferente para a elipse.\pause

			\item Se $(x,y)$ satisfaz a equação \eqref{equacaoelipse}, \pause então $(-x,-y)$, \pause $(-x,y)$ \pause e $(x,-y)$ \pause também satisfazem a
			      mesma equação. \pause Assim a elipse é simétrica em relação ao eixo $x$, \pause ao eixo $y$ \pause e ao centro do plano cartesiano.
			      \seti
		\end{enumerate}
	\end{observacao}
\end{frame}
\begin{frame}
	\begin{observacao}
		\begin{enumerate}[label=({\arabic*})]
			\conti
			\item O ponto $(x,0)$ pertence à elipse de equação \eqref{equacaoelipse} \pause se, e somente se, \pause $x^2/a^2 = 1$, \pause isto é, $x = \pm a$.
			      \pause Além disso, o ponto $(0,y)$ pertence à elipse de equação \eqref{equacaoelipse} \pause se, e só se, $y = \pm b$. \pause Portanto, a
			      interseção de $\mathcal{E}$ com o eixo $x$ \pause são os pontos $A_1(-a,0)$ e $A_2(a,0)$ \pause e a interseção com o eixo $y$ \pause são os
			      pontos $B_1(0,-b)$ e $B_2(0,b)$. \pause Note que $d(B_j,F_i) = \sqrt{c^2 + b^2} = a$, \pause para $i$, $j = 1$, 2.\pause

			\item Não existe circunferência que contenha os pontos $A_1$, $A_2$, \pause $B_1$ e $B_2$. \pause Caso existisse, \pause tal circunferência teria
			      dois diâmetros diferentes, \pause cujos comprimento são dados pelos comprimentos dos segmentod $A_1A_2$ \pause e $B_1B_2$, \pause o que é
			      impossível. \pause Portanto, como esses pontos pertencem a $\mathcal{E}$, \pause podemos afirmar que a elipse não é uma circunferência \pause e
			      nem o conjunto vazio.
		\end{enumerate}
	\end{observacao}
\end{frame}

	\end{proposicao}
\end{frame}

\begin{frame}
	\begin{corolario}

		\end{enumerate}
	\end{corolario}
\end{frame}

\begin{frame}
	\begin{definicao}
		Sejam $F_1$ e $F_2$ pontos distintos do plano tais que sua distância seja $2c$. Considere um número real $a$ tal que $0 < a < c$. O lugar geométrico $\mathcal{H}$ dos pontos $P(x,y)$ tais que
		\begin{equation}\label{equacaohiperbole}
			| d(P,F) - d(P,F_2)| = 2a
		\end{equation}
		chama-se \textbf{hipérbole}. Os pontos $F_1$ e $F_2$ são chamados \textbf{focos} da hipérbole, o segmento $F_1F_2$ é chamado \textbf{segmento focal}, o ponto médio do segmento $F_1F_2$ é chamada de \textbf{centro} da hipérbole e o número $2c$ é chamado de \textbf{distância focal}. A reta passando por $F_1$ e $F_2$ é chamada \textbf{reta focal}.\index{Hipérbole} \index{Hipérbole!Focos} \index{Hipérbole!Segmento Focal} \index{Hipérbole!Centro} \index{Hipérbole!Distância Focal} \index{Hipérbole!Reta Focal}
	\end{definicao}
\end{frame}

\begin{frame}
	\begin{proposicao}
		Um ponto $P(x, y)$ é um ponto da hipérbole
		\[
			\mathcal{H}:\ \dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1
		\]
		se, e somente se, as distâncias de $P$ aos focos são:
		\begin{align*}
			d(A,F_1) = \left|\dfrac{c}{a}\alpha + a\right| \\
			d(A,F_2) = \left|\dfrac{c}{a}\alpha - a\right|.
		\end{align*}
	\end{proposicao}

	Os números $a$, $b$ e $c$ são chamados \textbf{parâmetros geométricos} da hipérbole $\mathcal{H}$ e a equação
	\[
		\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1
	\]
	é chamada de \textbf{equação reduzida} da hipérbole.\index{Hipérbole!Parâmetros Geométricos} \index{Hipérbole!Equação Reduzida}
\end{frame}

\begin{frame}
	\begin{observacao}
		Seja $\mathcal{H} : \dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ uma hipérbole de focos no eixo $x$.
		\begin{enumerate}[label=({\arabic*})]
			\item Nenhum ponto $P(x,y)$ de $\mathcal{H}$ é tal que $-a < x < a$, pois caso contrário teríamos $x^2 < a^2$ o que contradiz a inequação \eqref{pontohiperbole}. Para a ordenada $y$ não há restrição, isto é, qualquer que seja $y$ existe $x$ tal que $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$. Assim a hipérbole não é limitada.
			\item Se $(x,y)$ pertence a $H$, então $(-x,-y)$, $(x,-y)$ e $(-x,y)$ também pertencem a $\mathcal{H}$. Logo a hipérbole é simétrica em relação ao eixo $x$, ao eixo $y$  e em relação á origem do plano cartesiano.
			      \seti
		\end{enumerate}
	\end{observacao}

\end{frame}

\begin{frame}
	\begin{observacao}
		\begin{enumerate}[label=({\arabic*})]
			\conti
			\item O ponto $(x,0)$ pertence á hipérbole $\mathcal{H}$ se, e s\'o se, $x = \pm a$. Por isso a interseção de $\mathcal{H}$ com a reta focal são os pontos $A_1(-a,0)$ e $A_2(a,0)$, chamados de \textbf{vértices} de $\mathcal{H}$.\index{Hipérbole!Vértices}
		\end{enumerate}
	\end{observacao}
\end{frame}

\begin{frame}
	\begin{proposicao}
		Um equação da forma
		\begin{equation}\label{hiperbolegeral}
			\dfrac{x^2}{p} + \dfrac{y^2}{q} = 1
		\end{equation}
		descreve uma hipérbole se, e somente se, os números reais $p$ e $q$ são de sinal contrário.
	\end{proposicao}
\end{frame}

\begin{frame}
	\begin{corolario}
		Sejam $p$ e $q$ números reais de sinal contrário e $\mathcal{H}$ a hipérbole de equação \eqref{hiperbolegeral} e parâmetros geométricos $a$ e $b$.
		\begin{enumerate}[label=({\arabic*})]
			\item Se $p > 0$ e $q < 0$, então $a^2 = p$ e $b^2 = -q$, $\mathcal{H}$ tem centro na origem, focos no eixo $x$ e assíntotas dadas por $y = \pm(b/a)x$.
			\item Se $p < 0$ e $q > 0$, então $a^2 = q$ e $b^2 = -p$, $\mathcal{H}$ tem centro na origem, focos no eixo $y$ e assíntotas dadas por $x = \pm(a/b)y$.
		\end{enumerate}
	\end{corolario}
\end{frame}

\begin{frame}

		Seja $r$ uma reta e $F$ um ponto que não pertence a $r$. O lugar geométrico $\mathcal{P}$ dos pontos equidistantes de $F$ e $r$ chama-se \textbf{parábola}. O ponto $F$ é chamado de \textbf{foco da parábola} e $r$ de \textbf{reta diretriz}.\index{Parábola} \index{Parábola!Focos} A reta contendo o foco $F$ e perpendicular á reta diretriz é chamada de \textbf{reta focal}.\index{Parábola!Reta diretriz}\index{Parábola!Reta focal}
	\end{definicao}
\end{frame}

\begin{frame}
	Para determinar a cônica representada pela equação
	\[
		g(x, y) = ax^2 + bxy + cy^2 + dx + ey + f = 0
	\]
	iremos utilizar translações e rotações de eixos para simplificar sua equação.
\end{frame}

\begin{frame}
	De modo geral, se $(x,y)$ são as coordenadas de um ponto $P$ em relação ao ponto $(0,0)$, então as coordenadas de $P$ em relação aos eixos $x_1$ e $y_1$, centrados no ponto $(\alpha,\beta)$ são dadas por\index{Translação de Eixos}
	\begin{align}
		x_1 = x - \alpha\label{mudancaXgeral} \\
		y_1 = y - \beta\label{mudancaYgeral}.
	\end{align}
\end{frame}

\begin{frame}
	Considere uma cônica de equação
	\begin{equation}\label{equacao_geral_conica}
		ax^2 + bxy + cx^2 + dx + ey + f = 0.
	\end{equation}

	Queremos remover o termo $xy$ da equação \eqref{equacao_geral_conica} para então usar uma translação de eixos, se necessário, e com isso identificar a cônica definida por tal equação. Para isso vamos tentar encontrar um ângulo $\theta$ tal que a equação \eqref{equacao_geral_conica} possa ser transformada numa equação da forma
	\begin{equation}\label{equacao_rotacionada_conica}
		a'x_1^2 + c'y_1^2 + d'x_1 + e'y_1 + f' = 0.
	\end{equation}
\end{frame}

\begin{frame}
	\begin{teorema}
		Considere a equação
		\[
			ax^2 + bxy + cy^2 + dx + ey + f = 0
		\]
		com $a$, $b$, $c$, $d$, $e$, $f \in \real$, sendo $a$, $b$ e $c$ não necessariamente nulos. Então mediante uma rotação da forma
		\[
			X = R_\theta X_1
		\]
		onde
		\[
			X = \begin{bmatrix}
				x \\y
			\end{bmatrix}, \quad
			R_\theta = \begin{bmatrix}
				\cos\theta & -\sin\theta \\
				\sin\theta & \cos\theta
			\end{bmatrix}, \quad\mbox{e}\quad
			X_1 = \begin{bmatrix}
				x_1 \\y_1
			\end{bmatrix}
		\]
	\end{teorema}
\end{frame}

\begin{frame}
	\begin{teorema}
		a equação da cônica acima pode ser transformada em
		\[
			a'x_1^2 + c'y_1^2 + d'x_1 + e'y_1 + f = 0
		\]
		em que $a'$, $c'$ são raízes de
		\[
			p(\lambda) = \det(A - \lambda I_2) = \det \begin{bmatrix}
				a - \lambda & b/2         \\
				b/2         & c - \lambda
			\end{bmatrix}.
		\]

		Mais ainda,
		\[
			U_1 = \begin{bmatrix}
				\cos\theta \\ \sin\theta
			\end{bmatrix}
		\]
		é uma solução de norma 1 do sistema linear
		\[
			\begin{bmatrix}
				a - a' & b/2    \\
				b/2    & c - a'
			\end{bmatrix} \begin{bmatrix}
				x \\y
			\end{bmatrix} = \begin{bmatrix}
				0 \\0
			\end{bmatrix}.
		\]
	\end{teorema}
\end{frame}

\begin{frame}
	\begin{teorema}
		Seja $\mathcal{C}$ o conjunto dos pontos do plano que satisfazem a equação
		\[
			ax^2 + bxy + cy^2 + dx + ey + f = 0,
		\]
		com $a$, $b$, $c$, $d$, $e$, $f \in \real$, sendo $a$, $b$ e $c$ não simultaneamente nulos. Sejam $a'$ e $c'$ raízes de
		\[
			p(\lambda) = \det \begin{bmatrix}
				a - \lambda & b/2         \\
				b/2         & c - \lambda
			\end{bmatrix}.
		\]
		\begin{enumerate}[label=({\roman*})]
			\item Temos $a'c' = ac - b^2/4$.
			\item Se $a'c' > 0$, então $\mathcal{C}$ é uma elipse, um ponto ou o conjunto vazio.
			\item Se $a'c' < 0$, então $\mathcal{C}$ é uma hipérbole ou um par de retas concorrentes.
			\item Se $a'c' = 0$, então $\mathcal{C}$ é uma parábola, um par de retas paralelas, uma reta ou o conjunto vazio.
		\end{enumerate}
	\end{teorema}
\end{frame}
