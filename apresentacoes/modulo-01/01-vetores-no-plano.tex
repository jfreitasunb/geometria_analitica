    \begin{frame}
        Seja $P$ um ponto qualquer do plano.\pause 

        Por $P$ podemos traçar uma única reta $r$ paralela à reta $x$.\pause

        E uma única reta $s$ paralela à reta $y$.\pause 

        Estas retas interceptam $x$ e $y$ em pontos $P_x$ \pause e $P_y$, respectivamente.\pause

        Denote por $\alpha$ o número real correspondente a $P_x$ \pause e por $\beta$ o número real correspondente a $P_y$.\pause

        Estes dois números determinam o ponto $P$, \pause isto é, \pause conhecendo $\alpha$ e $\beta$ \pause podemos traças as retas $r$ e $s$.\pause 

        O ponto $P$ é a interseção destas retas.\pause 

        Os números $\alpha$ e $\beta$ \pause são chamados \textbf{abscissa} \pause e \textbf{ordenada} do ponto $P$, \pause respectivamente.\pause

        Eles formam as \textbf{coordenadas} de $P$.
    \end{frame}

    \begin{frame}
        Assim o par ordenado $(\alpha, \beta)$ pode ser representado graficamente por um ponto ou por uma seta. \pause

        \vspace{.3cm}

        Quando representamos $(\alpha, \beta)$ por uma seta, \pause podemos associar a este para uma \textbf{direção}, \pause dada por uma reta passado por essa
        seta, \pause um \textbf{sentido} \pause, dado pela ponta da seta \pause e um \textbf{módulo},\pause que definiremos a seguir.\\

        \vspace{.3cm}

        O \textbf{módulo} do par $(\alpha, \beta)$ \pause será a distância do ponto $(\alpha, \beta)$ \pause até o ponto $O(0, 0)$ \pause e é dada por\pause
        \[
            \sqrt{(\alpha - 0)^2 + (\beta - 0)^2} = \pause \sqrt{\alpha^2 + \beta^2}\pause
        \]
        que é o comprimento da seta.
    \end{frame}

          Sejam $\vec{u}$, \pause $\vec{v}$ \pause e $\vec{w}$ \pause vetores quaisquer. \pause As seguintes propriedades são verdadeiras:\pause
          \begin{enumerate}[label=({\roman*})]
            \item $(\vec{u} + \vec{v}) \pause + \vec{w} \pause = \vec{u} + \pause (\vec{v} + \vec{w})$ \pause (Associatividade da soma)\pause
            \item $\vec{u} + \pause \vec{v} = \pause \vec{v} + \vec{u}$ \pause (Comutatividade da soma)\pause
            \item Existe um único vetor $\vec{w}$ \pause que somado a $\vec{u}$ \pause resulta no próprio vetor $\vec{u}$:\pause
            \[
              \vec{u} + \vec{w} \pause = \vec{w} + \vec{u} \pause = \vec{u}.\pause
            \]

            \item Para cada $\vec{u}$, \pause existe um único vetor $\vec{v}$ \pause tal que\pause

            \]
            é um vetor unitário na direção de $\vec{u}$ pois

    \begin{frame}
        \begin{definicao}
            Sejam $\vec{u}$ e $\vec{v}$ vetores não-nulos. Então $\vec{u} = \vec{v}$ se, e somente se, $\vec{u}$ e $\vec{v}$ têm normas iguais, são de mesma direção e de mesmo sentido.
        \end{definicao}

          Dois vetores não nulos $\vec{u}$ e $\vec{v}$ são paralelos se, e somente se, existe um escalar $\lambda$ tal que $\vec{u} = \lambda\vec{v}$ (consequentemente, $\lambda \ne 0$ e assim $\vec{v} = \vec{u}/\lambda$).
        \end{proposicao}
    \end{frame}

        temos
        \begin{align}\label{equacaonorma}
          \norm{\vec{u} - \vec{v}}^2 = \norm{\vec{u}}^2 + \norm{\vec{v}}^2 - 2\norm{\vec{u}}\norm{\vec{v}}\cos\theta.
        \end{align}
    \end{frame}

    \begin{frame}
        \begin{definicao}\label{produtointerno}
          O \textbf{produto escalar} ou \textbf{produto interno} dos vetores $\vec{u}$ e $\vec{v}$, indicado por $\inner{u}{v}$, é o número real tal que:\index{Vetores!Produto Escalar}
          \begin{enumerate}[label=({\roman*})]
            \item Se $\vec{u}$ ou $\vec{v}$ é nulo, então $\inner{u}{v} = 0$.
            \item Se $\vec{u}$ e $\vec{v}$ não são nulos e $\theta$ é o ângulo entre $\vec{u}$ e $\vec{v}$, então
            \begin{align}\label{produto-interno}
              \inner{u}{v} = \norm{\vec{u}}\norm{\vec{v}}\cos\theta.
            \end{align}
          \end{enumerate}
        \end{definicao}
    \end{frame}

    \begin{frame}
        \begin{proposicao}
          Sejam $\vec{u}$ e $\vec{v}$ vetores e $\theta$ o ângulo entre $\vec{u}$ e $\vec{v}$.
          \begin{enumerate}[label=({\roman*})]
            \item Se $\vec{u}$ e $\vec{v}$ não são nulos, então
            \[
              \cos\theta = \dfrac{\inner{u}{v}}{\norm{\vec{u}}\norm{\vec{v}}}.
            \]
            \item Qualquer que seja o vetor $\vec{u}$,
            \[
              \norm{\vec{u}} = \sqrt{\inner{u}{u}}.
            \]
            \item Quaisquer que sejam os vetores $\vec{u}$ e $\vec{v}$, $\vec{u}\perp\vec{v}$ se, e somente se, $\inner{u}{v} = 0$.
            \end{enumerate}
        \end{proposicao}
   \end{frame}

    \begin{frame}
        Agora, como podemos encontrar o produto interno sem necessitar de determinar o ângulo $\theta$ entre dois vetores $\vec{u}$ e $\vec{v}$\,? Substituindo o produto interno na equação \eqref{equacaonorma} temos
        \[
          \norm{\vec{u} - \vec{v}}^2 = \norm{\vec{u}}^2 + \norm{\vec{v}}^2 - 2\inner{u}{v}
        \]
        isto é,
        \begin{align}\label{equacao-auxilar-norma}
          \inner{u}{v} = \dfrac{1}{2}(\norm{\vec{u}}^2 + \norm{\vec{v}}^2 - \norm{\vec{u} - \vec{v}}^2).
        \end{align}
        Se $\vec{u} = (x_1, y_1)$ e $\vec{v} = (x_2, y_2)$, então podemos reescrever \eqref{equacao-auxilar-norma} como
        \begin{align*}
          \inner{u}{v} &= \dfrac{1}{2}(x_1^2 + y_1^2 + x_2^2 + y_2^2 - \norm{(x_1 - x_2, y_1 - y_2)}^2)\\
          &= \dfrac{1}{2}[x_1^2 + y_1^2 + x_2^2 + y_2^2 - (x_1 - x_2)^2 - (y_1 - y_2)^2]\\
          &= x_1x_2 + y_1y_2.
        \end{align*}
    \end{frame}

    \begin{frame}
        Acabamos de demonstrar o seguinte teorema:
        \begin{teorema}
          O produto interno de dois vetores $\vec{u} = (x_1, y_1)$ e $\vec{v} = (x_2, y_2)$ em $\real^2$ é dado por
          \[
            \inner{u}{v} = x_1x_2 + y_1y_2.
          \]
        \end{teorema}
    \end{frame}

    \begin{frame}
        \begin{exemplos}
          \begin{enumerate}[label=({\roman*})]
            \item Sejam $\vec{u} = (1, 1)$ e $\vec{v} = (1,0)$. Determine o ângulo entre $\vec{u}$ e $\vec{v}$.
            \item Encontre um vetor $\vec{u} = (x, y)$ ortogonal a $\vec{v} = (4, -1)$ e tal que $\inner{u}{w} = -1$, onde $\vec{w} = (1,1)$.
          \end{enumerate}
        \end{exemplos}
    \end{frame}

    \begin{frame}
        \begin{proposicao}\label{propriedades-produto-interno}
          Quaisquer que sejam os vetores $\vec{u} = (x_1, y_1)$, $\vec{v} = (x_2, y_2)$ e $\vec{w} = (x_3, y_3)$ e qualquer que seja o número real $\lambda$ temos
          \begin{enumerate}[label=({\roman*})]
            \item\label{linearidade-produto-interno} $\langle \vec{u}, \vec{v} + \vec{w}\rangle = \inner{u}{v} + \inner{u}{w}$
            \item $\langle \vec{u} + \vec{v}, \vec{w}\rangle = \inner{u}{w} + \inner{v}{w}$
            \item\label{linearidade2-produto-interno} $\langle (\lambda \vec{u}\,), \vec{v}\rangle = \langle \vec{u}, (\lambda \vec{v}\,)\rangle = \lambda\inner{u}{v}$
            \item $\inner{u}{v} = \inner{v}{u}$
            \item Se $\vec{u} \ne \vec{0}$, então $\inner{u}{u} > 0$.
          \end{enumerate}
        \end{proposicao}
   \end{frame}

    \begin{frame}
        \begin{observacao}
          \begin{enumerate}[label=({\roman*})]
            \item As propriedades \ref{linearidade-produto-interno} e \ref{linearidade2-produto-interno} Proposição \ref{propriedades-produto-interno} podem ser estendidas para qualquer número de vetores e escalares:
            \[
              \langle\vec{u}, (\lambda_1\vec{v_1} + \lambda_2\vec{v_2} + \cdots + \lambda_n\vec{v_n} )\rangle = \lambda_1\inner{u}{v_1} + \lambda_2\inner{u}{v_2} + \cdots + \lambda_n\inner{u}{v_n}.
            \]
            \item Na igualdade $\inner{u}{v} = \inner{u}{w}$ não podemos concluir que $\vec{v} = \vec{w}$. Por exemplo, para $\vec{u} = (1, 0)$, $\vec{v} = (2, 1)$ e $\vec{w} = (2, -5)$ temos $\inner{u}{v} = \inner{u}{w}$ e no entanto $\vec{v} \ne \vec{w}$. Mas podemos concluir que $\vec{u}\perp(\vec{v} - \vec{w})$.
            \item De $\inner{u}{v} = 0$ não podemos concluir que $\vec{u} = \vec{0}$ ou $\vec{v} = \vec{0}$. Por exemplo, para $\vec{u} = (2, 1)$ e $\vec{v} = (-2, 4)$ temos $\inner{u}{v} = 0$ e no entanto $\vec{u} \ne \vec{0}$ e $\vec{v} \ne \vec{0}$.
          \end{enumerate}
        \end{observacao}
    \end{frame}

    \begin{frame}


        Sejam $\vec{u} = (x_1, y_1)$ e $\vec{v} = (x_2, y_2)$ vetores não nulos. Suponha que o ângulo $\theta$ entre $\vec{u}$ e $\vec{v}$ é tal que $0 < \theta < \pi/2$.
        \begin{figure}[!h]
          \centering
          \caption{Projeção ortogonal}
          \includegraphics{projecao-vetores-plano-cartesiano.pdf}

        O vetor $\vec{OP}$ é paralelo ao vetor $\vec{u}$ e o vetor $\vec{PA}$ é ortogonal ao vetor $\vec{u}$.
    \end{frame}

    \begin{frame}
        \begin{definicao}\label{projecao_ortogonal}
          Seja $\vec{u}$ um vetor não nulo. Dado qualquer vetor $\vec{v}$, o vetor $\vec{OP}$ é chamado de \textbf{projeção ortogonal} de $\vec{v}$ sobre $\vec{u}$, e é indicado por $\proj_{\vec{u}}\vec{v}$ se satisfaz as seguintes condições:\index{Vetores!Projeção ortogonal}
          \begin{enumerate}[label=({\roman*})]
            \item $\vec{OP} \varparallel\vec{u}$
            \item $(\vec{v} - \vec{OP}) \perp \vec{u}$.
          \end{enumerate}
        \end{definicao}
    \end{frame}

    \begin{frame}
        Podemos também escrever
        \begin{enumerate}[label=({\roman*})]
          \item $\proj_{\vec{u}}\vec{v} \varparallel\vec{u}$
            \item $(\vec{v} - \proj_{\vec{u}}\vec{v}) \perp \vec{u}$.
        \end{enumerate}

        No caso acima temos
        \[
          \cos\theta = \dfrac{\norm{\vec{OP}}}{\norm{\vec{v}}}
        \]
        e então $\norm{\vec{OP}} = \norm{\vec{v}}\cos\theta$. Mas
        \[
          \cos\theta = \dfrac{\inner{u}{v}}{\norm{\vec{u}}\norm{\vec{v}}}
        \]
        e assim podemos escrever
        \[
          \norm{\vec{OP}} = \dfrac{\inner{u}{v}}{\norm{\vec{u}}}.
        \]
    \end{frame}

    \begin{frame}
        Além disso, como $\vec{OP}$ é paralelo à $\vec{u}$ e de mesmo sentido, devemos ter $\vec{OP} = \lambda\vec{u}$ com $\lambda > 0$. Daí
        $\lambda = \norm{\vec{OP}}/\norm{\vec{u}}$ e então
        \[
          \vec{OP} = \dfrac{\norm{OP}}{\norm{\vec{u}}^2}\vec{u}.
        \]
    \end{frame}

    \begin{frame}
        Esta construção só é válida no caso em que o ângulo $\theta$ entre $\vec{u}$ e $\vec{v}$ satisfaz $0 < \theta < \pi/2$. Nos casos
        \begin{figure}
          \centering
          \caption{Vetor projeção}
          \includegraphics[scale=0.8]{vetores-projecao-plano-cartesiano.pdf}
        \end{figure}
        não podemos usar a construção envolvendo o ângulo $\theta$ entre os vetores.
    \end{frame}

    \begin{frame}
        \begin{proposicao}
          Seja $\vec{u}$ um vetor não nulo. Qualquer que seja o vetor $\vec{v}$, existe e é única a projeção ortogonal de $\vec{v}$ sobre $\vec{u}$. Sua expressão em termos de $\vec{u}$ e $\vec{v}$ é dada por
          \[
            \proj_{\vec{u}}\vec{v} = \dfrac{\inner{v}{u}}{\norm{\vec{u}}^2}\vec{u}
          \]
          e sua norma é dada por
          \[
            \norm{\proj_{\vec{u}}\vec{v}} = \dfrac{\mid\inner{v}{u}\mid}{\norm{\vec{u}}}.
          \]
        \end{proposicao}
    \end{frame}

    \begin{frame}
        \begin{observacao}
          O escalar $\lambda$ encontrado na proposição anterior pode ser negativo ou zero. Ele será negativo quando $\pi/2 \le \theta \le \pi$ e será zero quando $\vec{u}\perp\vec{v}$.
          \begin{figure}[!h]
            \centering
            \caption{Possibilidades para $\lambda$}
            \includegraphics{vetores-projecao-escalar-plano-cartesiano.pdf}
          \end{figure}
        \end{observacao}
    \end{frame}

        \begin{exemplos}\label{exemplosprojecao}

            \item Encontre a projeção ortogonal do vetor $\vec{v} = (4, -1)$ sobre o vetor $\vec{u} = (2, 1)$.
             \item Considere um triângulo $ABC$ onde $A(3,3)$, $B(0,1)$ e $C(1,6)$.
             \begin{enumerate}[label=({\roman*})]\label{exemploprojecao2}

               \item Calcule a projeção do cateto $BA$ sobre a hipotenusa $BC$.
               \item Determine o pé da altura do triângulo relativo ao vértice $A$.
             \end{enumerate}
          \end{enumerate}
        \end{exemplos}
    \end{frame}
