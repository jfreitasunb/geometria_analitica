\begin{frame}
    Vamos considerar o plano definido pelo par de retas perpendiculares $x$ e $y$ como abaixo:
    \begin{figure}[!h]
        \centering
        \caption{Plano cartesiano}
        \includegraphics{plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    Seja $P$ um ponto qualquer do plano.\pause

    Por $P$ podemos traçar uma única reta $r$ paralela à reta $x$.\pause

    E uma única reta $s$ paralela à reta $y$.\pause

    Estas retas interceptam $x$ e $y$ em pontos $P_x$ \pause e $P_y$, respectivamente.\pause

    Denote por $\alpha$ o número real correspondente a $P_x$ \pause e por $\beta$ o número real correspondente a $P_y$.\pause

    Estes dois números determinam o ponto $P$, \pause isto é, \pause conhecendo $\alpha$ e $\beta$ \pause podemos traças as retas $r$ e $s$.\pause

    O ponto $P$ é a interseção destas retas.\pause

    Os números $\alpha$ e $\beta$ \pause são chamados \textbf{abscissa} \pause e \textbf{ordenada} do ponto $P$, \pause respectivamente.\pause

    Eles formam as \textbf{coordenadas} de $P$.
\end{frame}

\begin{frame}
    \begin{figure}[!h]
        \centering
        \caption{Coordenadas do ponto $P$}
        \includegraphics{coordenadas-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}

    Denotaremos assim o ponto $P$ por
    \[
        P(\alpha,\beta).
    \]
\end{frame}

\begin{frame}

    Sejam $P(x_1,y_1)$ e $Q(x_2,y_2)$ pontos do plano. \pause

    A partir de $P$ e $Q$ podemos construir o triângulo retângulo $PQS$ \pause como abaixo:
    \begin{figure}[!h]
        \centering
        \caption{Distância entre dois pontos}
        \includegraphics{distancia-pontos-plano-cartesiano.pdf}
    \end{figure}\pause
\end{frame}

\begin{frame}
    Este número é chamado \textbf{distância} de $P$ a $Q$ e é indicado por $d(P,Q)$, isto é, por definição
    \[
        d(P,Q) = \sqrt{(x_1 - x_2)^2 + (y_1 - y_2)^2}.\pause
    \]

    Dizemos que dois pontos $P(x_1,y_1)$ e $Q(x_2,y_2)$ \pause são iguais \pause se, e somente se, $x_1 = x_2$ \pause e $y_1 = y_2$.

\end{frame}

\begin{frame}
    Sabemos que a cada par ordenado $(\alpha, \beta)$ corresponde um ponto do plano.\pause
    \vspace{.3cm}

    Assim se $(\alpha, \beta) \ne (0,0)$ \pause podemos associar uma seta \pause com início no ponto $O(0,0)$ \pause e final no ponto $(\alpha,\beta)$
    \pause como na figura abaixo:\pause

    \begin{figure}[!h]
        \centering
        \caption{Seta representando o ponto $(\alpha, \beta)$.}
        \includegraphics{vetores-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    Assim o par ordenado $(\alpha, \beta)$ pode ser representado graficamente por um ponto ou por uma seta. \pause

    \vspace{.3cm}

    Quando representamos $(\alpha, \beta)$ por uma seta, \pause podemos associar a este para uma \textbf{direção}, \pause dada por uma reta passado por essa
    seta, \pause um \textbf{sentido} \pause, dado pela ponta da seta \pause e um \textbf{módulo},\pause que definiremos a seguir.\\

    \vspace{.3cm}

    O \textbf{módulo} do par $(\alpha, \beta)$ \pause será a distância do ponto $(\alpha, \beta)$ \pause até o ponto $O(0, 0)$ \pause e é dada por\pause
    \[
        \sqrt{(\alpha - 0)^2 + (\beta - 0)^2} = \pause \sqrt{\alpha^2 + \beta^2}\pause
    \]
    que é o comprimento da seta.
\end{frame}

\begin{frame}
    Em geral, um objeto ao qual podemos associar os conceitos de direção, \pause sentido \pause e módulo \pause é chamado de \textbf{vetor}.\pause\\

    \vspace{.3cm}

    Denotaremos um vetor de extremidade em um ponto $(\alpha, \beta)$ \pause por
    \[
        \vec{u} = (\alpha, \beta).
    \]
\end{frame}

\begin{frame}
    Esta notação significa que $\vec{u}$ é a seta com início no ponto $O(0,0)$ \pause e final no ponto $(\alpha, \beta)$.\pause
    \begin{figure}[!h]
        \centering
        \caption{Vetor $ \vec{u} = (\alpha, \beta)$.}
        \includegraphics{vetor-coordenadas-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    Por exemplo, $\vec{u} = (3,2)$ \pause e $\vec{v} = (-1,-2)$ \pause são representados graficamente por:\pause
    \begin{figure}[!h]
        \centering
        \caption{Vetores $\vec{u} = (3,2)$ e $\vec{v} = (-1,-2)$.}
        \includegraphics{exemplo-vetores-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    O vetor que tem início e fim no ponto $(0,0)$ \pause é chamado de \textbf{vetor nulo} \pause e será denotado por\pause
    \[
        \vec{O} = (0,0).
    \]
\end{frame}

    vetor\pause

        \vec{AB} = (x_2 - x_1, y_2 - y_1).

\begin{frame}

    \begin{definicao}
        Seja $\vec{u} = (x_1, y_1)$ um vetor. \pause A \textbf{norma} ou \textbf{módulo} de $\vec{u}$, \pause denotada por $\norm{\vec{u}}$, \pause é dada
        por\pause
        \[
            \norm{\vec{u}} = \pause \sqrt{x_1^2 + y_1^2}.
        \]
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\vec{u}$ e $\vec{v}$ vetores não-nulos. Então $\vec{u} = \vec{v}$ se, e somente se, $\vec{u}$ e $\vec{v}$ têm normas iguais, são de mesma
        direção e de mesmo sentido.
    \end{definicao}
\end{frame}

\begin{frame}
    Um vetor $\vec{u}$ \pause tal que
    \[
        \norm{\vec{u}} = 1\pause
    \]
    é chamado de \pause \textbf{vetor unitário}.\pause
    \begin{figure}[!h]
        \centering
        \caption{Vetores unitários: $\norm{\vec{u}} = 1$}
        \includegraphics{vetores-unitarios-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\vec{u} = (x_1, y_1)$ \pause e $\vec{v} = (x_2, y_2)$ \pause vetores. \pause Então:\pause
        \[
            \vec{u} + \vec{v} = \pause (x_1 + x_2, \pause y_1 + y_2).
        \]
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{figure}[!h]
        \centering
        \caption{Soma de vetores}
        \includegraphics{soma-vetores-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    \begin{propriedades}
        Sejam $\vec{u}$, \pause $\vec{v}$ \pause e $\vec{w}$ \pause vetores quaisquer. \pause As seguintes propriedades são verdadeiras:\pause
        \begin{enumerate}[label=({\roman*})]
            \item $(\vec{u} + \vec{v}) \pause + \vec{w} \pause = \vec{u} + \pause (\vec{v} + \vec{w})$ \pause (Associatividade da soma)\pause
            \item $\vec{u} + \pause \vec{v} = \pause \vec{v} + \vec{u}$ \pause (Comutatividade da soma)\pause
            \item Existe um único vetor $\vec{w}$ \pause que somado a $\vec{u}$ \pause resulta no próprio vetor $\vec{u}$:\pause
                  \[
                      \vec{u} + \vec{w} \pause = \vec{w} + \vec{u} \pause = \vec{u}.\pause
                  \]
                  Tal vetor $\vec{w}$ é o vetor nulo $\vec{0}$. \pause (Elemento Neutro da soma)\pause
            \item Para cada $\vec{u}$, \pause existe um único vetor $\vec{v}$ \pause tal que\pause
                  \[
                      \vec{u} + \vec{v} \pause = \vec{0} \pause = \vec{v} + \vec{u}.\pause
                  \]
        \end{enumerate}
    \end{propriedades}
\end{frame}

\begin{frame}
    \begin{definicao}
        O vetor $\vec{v}$ do item \textit{(iv)} \pause da definição anterior \pause é chamado de \textbf{vetor oposto}.
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\alpha \in \real$ e \pause $\vec{u} = (x_1, y_1)$ \pause um vetor. \pause Então:\pause
        \[
            \alpha \vec{u} \pause = (\alpha x_1, \pause \alpha y_1).
        \]
        Tal multiplicação é chamada de \pause \textbf{multiplicação por escalar} \pause e o número real $\alpha$ \pause é chamado de \pause \textbf{escalar}.
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{figure}[!h]
        \centering
        \caption{Multiplicação por escalar}
        \includegraphics{multiplicacao-escalar-vetores-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    \begin{observacao}
        \begin{enumerate}[label=({\roman*})]
            \item Sejam $\vec{u}$ um vetor não-nulo \pause e $\lambda \ne 0$ \pause um escalar. \pause
                  Além disso, \pause  se $\lambda > 0$, \pause então $\vec{u}$ e $\lambda\vec{u}$ têm o mesmo sentido, \pause e se $\lambda < 0$, \pause
                  então $\vec{u}$ e $\lambda\vec{u}$ têm sentidos contrários.\pause

                  \vspace{.3cm}

            \item Quando $\lambda = -1$, \pause o vetor $-\vec{u} = (-x_1,-y_1)$ \pause é exatamente o vetor oposto de $\vec{u}$. \pause Note que $-\vec{u}$
                  \pause tem a mesma direção que $\vec{u}$, \pause mas sentido contrário.
        \end{enumerate}
    \end{observacao}
\end{frame}

\begin{frame}
    \begin{propriedades}
        Quaisquer que sejam os números reais $\alpha$ \pause e $\beta$ \pause e quaisquer que sejam os vetores $\vec{u}$ \pause e $\vec{v}$ \pause valem as
        igualdades:\pause
        \begin{enumerate}[label=({\roman*})]
            \item $\alpha \pause (\vec{u} + \vec{v}) \pause = \alpha\vec{u} \pause + \alpha\vec{v}$;\pause
            \item $(\alpha + \beta)\pause \vec{u} =\pause \alpha\vec{u} \pause + \beta\vec{u}$;\pause
            \item $1\vec{u} \pause = \vec{u}$;\pause
            \item $\alpha \pause (\beta\pause \vec{u}) \pause = (\alpha\beta)\pause \vec{u} = \pause \beta\pause (\alpha\vec{u})$.
        \end{enumerate}
    \end{propriedades}
\end{frame}

\begin{frame}
    \begin{observacao}
        Dado um vetor não nulo $\vec{u}$ \pause o vetor
        \[
            \vec{v} = \pause \dfrac{\vec{u}}{\norm{\vec{u}}}\pause
        \]
        tem a mesma direção de $\vec{u}$ \pause e é unitário.
    \end{observacao}
\end{frame}

\begin{frame}
    \begin{proposicao}
        Sejam $\vec{u} = (x_1, y_1)$ \pause um vetor em $\real^2$ \pause e $\alpha \in \real$. \pause Então:\pause
        \begin{enumerate}[label=({\roman*})]
            \item Se $\vec{u} \ne \vec{0}$, \pause então $\norm{\vec{u}} \ne 0$.\pause
            \item $\norm{\vec{u}} = 0$ \pause se, e somente se, $\pause \vec{u} = \vec{0}$.\pause
            \item $\norm{\alpha\vec{u}} \pause = |\alpha|\pause \norm{\vec{u}}$.
        \end{enumerate}
    \end{proposicao}
\end{frame}

\begin{frame}
    \begin{definicao}
        Dois vetores $\vec{u}$ \pause e $\vec{v}$ \pause são \textbf{paralelos} \pause (ou \textbf{colineares}) \pause se as retas que contém estes vetores
        \pause são paralelas (podendo ser coincidentes). \pause Caso $\vec{u}$ e $\vec{v}$ sejam paralelos, \pause escreveremos $\vec{u}\varparallel\vec{v}$.
        \pause Neste caso, $\vec{u}$ e $\vec{v}$ \pause têm a mesma direção, mas podem ter sentidos contrários.\pause
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{figure}[!h]
        \centering
        \caption{Vetores paralelos}
        \includegraphics[scale=0.8]{vetores-paralelos-plano-cartesiano.pdf}
    \end{figure}

\begin{frame}
    \begin{proposicao}\label{vetores_paralelos}
        Dois vetores não nulos $\vec{u}$ e $\vec{v}$ \pause são paralelos \pause se, e somente se, \pause existe um escalar $\lambda$ \pause tal que $\vec{u}
            = \lambda\vec{v}$ \pause (consequentemente, $\lambda \ne 0$ \pause e assim $\vec{v} = \vec{u}/\lambda$).
    \end{proposicao}
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\vec{u}$ e $\vec{v}$ vetores não nulos. \pause O \textbf{ângulo} entre $\vec{u}$ e $\vec{v}$ \pause é definido pelo ângulo $\theta$ \pause
        determinado por $\vec{u}$ e $\vec{v}$ \pause e que satisfaz $0 \le \theta \le \pi$, \pause quando os vetores $\vec{u}$ e $\vec{v}$ são representados com a mesma origem.
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{figure}[!h]
        \centering
        \caption{ângulo entre vetores}
        \includegraphics{angulo-vetores-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

    \end{figure}
    \begin{align}\label{equacaonorma}
        \norm{\vec{u} - \vec{v}}^2 = \norm{\vec{u}}^2 + \norm{\vec{v}}^2 - 2\norm{\vec{u}}\norm{\vec{v}}\cos\theta.
    \end{align}
\end{frame}

\begin{frame}
    \begin{definicao}
        Quando o ângulo $\theta$ entre dois vetores $\vec{u}$ e $\vec{v}$ \pause é reto, isto é, $\theta = \pi/2$, \pause ou um deles é o vetor nulo, \pause
        dizemos que os vetores $\vec{u}$ e $\vec{v}$ \pause são \textbf{ortogonais} \pause ou \textbf{perpendiculares} entre si. \pause Denotamos tal fato, escrevendo $\vec{u} \perp \vec{v}$.
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{definicao}\label{produtointerno}
        O \textbf{produto escalar} \pause ou \textbf{produto interno} \pause dos vetores $\vec{u}$ e $\vec{v}$, \pause indicado por $\inner{u}{v}$, \pause é o
        número real tal que:\pause
        \begin{enumerate}[label=({\roman*})]
            \item Se $\vec{u}$ ou $\vec{v}$ é nulo, \pause então $\inner{u}{v} = 0$.\pause
            \item Se $\vec{u}$ e $\vec{v}$ não são nulos \pause e $\theta$ é o ângulo entre $\vec{u}$ e $\vec{v}$, \pause então\pause
                  \begin{align*}
                      \inner{u}{v} = \norm{\vec{u}}\norm{\vec{v}}\cos\theta.
                  \end{align*}
        \end{enumerate}
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{proposicao}
        Sejam $\vec{u}$ e $\vec{v}$ vetores \pause e $\theta$ o ângulo entre $\vec{u}$ e $\vec{v}$.\pause
        \begin{enumerate}[label=({\roman*})]
            \item Se $\vec{u}$ e $\vec{v}$ não são nulos, \pause então\pause
                  \[
                      \cos\theta = \dfrac{\inner{u}{v}}{\norm{\vec{u}}\norm{\vec{v}}}.\pause
                  \]
            \item Qualquer que seja o vetor $\vec{u}$,\pause
                  \[
                      \norm{\vec{u}} = \sqrt{\inner{u}{u}}.\pause
                  \]
            \item Quaisquer que sejam os vetores $\vec{u}$ e $\vec{v}$, \pause $\vec{u}\perp\vec{v}$ \pause se, e somente se, $\inner{u}{v} = 0$.\pause
        \end{enumerate}
    \end{proposicao}
\end{frame}

\begin{frame}
    \begin{teorema}
        O produto interno de dois vetores $\vec{u} = (x_1, y_1)$ \pause e $\vec{v} = (x_2, y_2)$ \pause em $\real^2$ é dado por\pause
        \[
            \inner{u}{v} = x_1x_2 + y_1y_2.
        \]
    \end{teorema}
\end{frame}

\begin{frame}
    \begin{proposicao}
        Quaisquer que sejam os vetores $\vec{u} = (x_1, y_1)$, \pause $\vec{v} = (x_2, y_2)$ \pause e $\vec{w} = (x_3, y_3)$ \pause e qualquer que seja o
        número real $\lambda$ \pause temos\pause
        \begin{enumerate}[label=({\roman*})]
            \item $\langle \vec{u}, \pause \vec{v} + \vec{w}\rangle \pause = \inner{u}{v} \pause + \inner{u}{w}$\pause
            \item $\langle \vec{u} + \vec{v}, \pause \vec{w}\rangle \pause = \inner{u}{w} \pause + \inner{v}{w}$\pause
            \item $\langle (\lambda \vec{u}\,), \vec{v}\rangle \pause = \langle \vec{u}, (\lambda \vec{v}\,)\rangle \pause = \lambda\inner{u}{v}$\pause
            \item $\inner{u}{v} = \pause \inner{v}{u}$\pause
            \item Se $\vec{u} \ne \vec{0}$, \pause então $\inner{u}{u} > 0$.
        \end{enumerate}
    \end{proposicao}
\end{frame}

\begin{frame}
    \begin{observacao}
        \begin{enumerate}[label=({\roman*})]
            \item As propriedades do produto interno \pause podem ser estendidas para qualquer número de vetores \pause e escalares:\pause
                  \[
                      \langle\vec{u}, \pause (\lambda_1\vec{v_1} + \pause \lambda_2\vec{v_2} \pause + \cdots + \pause \lambda_n\vec{v_n} )\rangle \pause =
                      \lambda_1\inner{u}{v_1} \pause + \lambda_2\inner{u}{v_2} \pause + \cdots + \pause \lambda_n\inner{u}{v_n}.\pause
                  \]
            \item Na igualdade $\inner{u}{v} = \inner{u}{w}$ \pause não podemos concluir que $\vec{v} = \vec{w}$. \pause Por exemplo, para $\vec{u} = (1, 0)$,
                  \pause $\vec{v} = (2, 1)$ \pause e $\vec{w} = (2, -5)$ \pause temos $\inner{u}{v} = \inner{u}{w}$ \pause e no entanto $\vec{v} \ne \vec{w}$.
                  \pause Mas podemos concluir que $\vec{u}\perp(\vec{v} - \vec{w})$.\pause
            \item De $\inner{u}{v} = 0$ \pause não podemos concluir que $\vec{u} = \vec{0}$ \pause ou $\vec{v} = \vec{0}$. \pause Por exemplo, para $\vec{u} =
                      (2, 1)$ \pause e $\vec{v} = (-2, 4)$ \pause temos $\inner{u}{v} = 0$ \pause e no entanto $\vec{u} \ne \vec{0}$ \pause e $\vec{v} \ne \vec{0}$.
        \end{enumerate}
    \end{observacao}
\end{frame}

\begin{frame}
    \begin{proposicao}
        Sejam $\vec{u}$ e $\vec{v}$ vetores.\pause
        \begin{enumerate}[label=({\roman*})]
            \item $|\inner{u}{v}| \pause \le \norm{\vec{u}}\pause \norm{\vec{v}}$ \pause (Desigualdade de Scharwz)\pause
            \item $\norm{\vec{u} + \vec{v}} \pause \le \norm{\vec{u}} + \pause \norm{\vec{v}}$ \pause (Desigualdade Triangular)
        \end{enumerate}
    \end{proposicao}
\end{frame}

\begin{frame}
    Sejam $\vec{u} = (x_1, y_1)$ e $\vec{v} = (x_2, y_2)$ vetores não nulos. \pause Suponha que o ângulo $\theta$ \pause entre $\vec{u}$ e $\vec{v}$ \pause
    é tal que $0 < \theta < \pi/2$.\pause
    \begin{figure}[!h]
        \centering
        \caption{Projeção ortogonal}
        \includegraphics{projecao-vetores-plano-cartesiano.pdf}\pause
    \end{figure}

    O vetor $\vec{OP}$ \pause é paralelo ao vetor $\vec{u}$ \pause e o vetor $\vec{PA}$ \pause é ortogonal ao vetor $\vec{u}$.\pause
\end{frame}

\begin{frame}
    \begin{definicao}
        Seja $\vec{u}$ um vetor não nulo. \pause Dado qualquer vetor $\vec{v}$, \pause o vetor $\vec{OP}$ \pause é chamado de \textbf{projeção ortogonal}
        \pause de $\vec{v}$ sobre $\vec{u}$, \pause e é indicado por $\proj_{\vec{u}}\vec{v}$ \pause se satisfaz as seguintes condições:\pause
        \begin{enumerate}[label=({\roman*})]
            \item $\vec{OP} \varparallel\vec{u}$\pause
            \item $(\vec{v} - \vec{OP}) \perp \vec{u}$.
        \end{enumerate}
    \end{definicao}
\end{frame}

\begin{frame}
    Podemos também escrever\pause
    \begin{enumerate}[label=({\roman*})]
        \item $\proj_{\vec{u}}\vec{v} \varparallel\vec{u}$\pause
        \item $(\vec{v} - \proj_{\vec{u}}\vec{v}) \perp \vec{u}$.\pause
    \end{enumerate}
\end{frame}

\begin{frame}
    Esta construção só é válida no caso em que o ângulo $\theta$ entre $\vec{u}$ e $\vec{v}$ satisfaz $0 < \theta < \pi/2$. Nos casos
    \begin{figure}
        \centering
        \caption{Vetor projeção}
        \includegraphics[scale=0.8]{vetores-projecao-plano-cartesiano.pdf}
    \end{figure}
    não podemos usar a construção envolvendo o ângulo $\theta$ entre os vetores.
\end{frame}

\begin{frame}
    \begin{proposicao}
        Seja $\vec{u}$ um vetor não nulo. \pause Qualquer que seja o vetor $\vec{v}$, \pause existe e é única \pause a projeção ortogonal de $\vec{v}$ sobre
        $\vec{u}$. \pause Sua expressão em termos de $\vec{u}$ e $\vec{v}$ \pause é dada por\pause
        \[
            \proj_{\vec{u}}\vec{v} = \pause \dfrac{\inner{v}{u}}{\norm{\vec{u}}^2}\vec{u}\pause
        \]
        e sua norma é dada por\pause
        \[
            \norm{\proj_{\vec{u}}\vec{v}} = \pause \dfrac{\mid\inner{v}{u}\mid}{\norm{\vec{u}}}.\pause
        \]
    \end{proposicao}
\end{frame}
