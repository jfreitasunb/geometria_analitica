\begin{frame}
    \begin{definicao}
        Qualquer vetor não nulo \pause e paralelo \pause a uma dada reta $r$ \pause é chamado de \textbf{vetor diretor} de $r$.
    \end{definicao}
\end{frame}

\begin{frame}
    Seja $\vec{u}$ um vetor diretor de uma reta $r$ \pause e seja $A$ um ponto pertencente à reta $r$. \pause Assim um ponto $X$ \pause pertence a reta $r$
    se, \pause e somente se, \pause $\vec{AX}\varparallel\vec{u}$.\pause
    \begin{figure}[!h]
        \centering
        \caption{Equação vetorial da reta}
        \includegraphics{equacao-vetorial-reta-plano-cartesiano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    \begin{definicao}\index{Reta no Plano!Equação Vetorial}
        A equação\pause
        \[
            \vec{AX} = \lambda \vec{u},\pause
        \]
        onde $\lambda \in \real$ \pause é chamada de \textbf{equação vetorial \pause da reta} $r$ \pause ou \textbf{equação da reta} $r$ \pause
        \textbf{na forma vetorial}.
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{observacao}
        Como o ponto \pause e o vetor diretor \pause de uma reta $r$ \pause são escolhidos de forma arbitrária, \pause existem infinitas equações vetoriais para uma
        mesma reta $r$. \pause Por exemplo, \pause se $A$ e $B$ são pontos distintos de uma reta $r$, \pause então $\vec{AB}$ \pause e $\vec{BA}$ \pause são vetores diretores distintos de $r$ e portanto temos as equações vetoriais\pause
        \begin{align*}
            \vec{AX} = \lambda\vec{AB} \\
            \vec{AX} = \alpha\vec{BA}.
        \end{align*}
        Na verdade, \pause qualquer vetor múltiplo do vetor $\vec{AB}$ \pause pode ser tomado como vetor diretor de $r$.
    \end{observacao}
\end{frame}

\begin{frame}
    \begin{definicao}
        Seja $r$ uma reta \pause com vetor diretor $\vec{u} = (a,b)$, \pause e que passa pelo ponto $(x_0, y_0)$. \pause O sistema de equações \pause
        \[
            \begin{cases}
                x = x_0 + \lambda a \\
                y = y_0 + \lambda b
            \end{cases}, \lambda \in \real
        \]\pause

        é chamado \textbf{sistema de equações paramétricas da reta} $r$ \pause ou \textbf{sistema de equações da reta} \pause $r$ \textbf{na forma paramétrica}.
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{definicao}
        Seja $r$ uma reta \pause com vetor diretor $\vec{u} = (a,b)$, \pause onde $ab\ne 0$ \pause e que passa pelo ponto $(x_0, y_0)$. \pause A equação\pause
        \begin{equation*}
            \dfrac{x - x_0}{a} = \dfrac{y - y_0}{b}\pause
        \end{equation*}
        é chamada de \pause \textbf{sistema de equações da reta} $r$ \pause \textbf{na forma simétrica} \pause ou por abuso de linguagem, \pause
        \textbf{equações da reta} $r$ \pause\textbf{na forma simétrica}.
    \end{definicao}
\end{frame}

\begin{frame}
    \begin{definicao}
        Seja $r$ uma reta que contém o ponto $(x_0, y_0)$ \pause e com vetor diretor $\vec{u} = (a, b)$. \pause A equação \pause
        \begin{equation*}
            ay - bx = c\pause
        \end{equation*}
        onde $c = ay_0 - bx_0$ \pause é chamada de \pause \textbf{equação cartesiana da reta} $r$.
    \end{definicao}
\end{frame}
