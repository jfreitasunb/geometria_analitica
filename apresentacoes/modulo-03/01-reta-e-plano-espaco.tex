\begin{frame}
    Da Geometria Plana, \pause sabemos que todo plano $\pi$ \pause contém pelo menos três pontos não colineares, \pause digamos $A$, \pause $B$ e $C$. \pause À
    partir desses pontos podemos construir os vetores:\pause
    \begin{align*}
        \vec{u} = \vec{AB} \\
        \vec{v} = \vec{AC}.
    \end{align*}
    Como $\vec{u}$ e $\vec{v}$ não são paralelos, \pause então podemos considerar o vetor $\vec{u}\times\vec{v}\ne \vec{0}$.\pause
\end{frame}

\begin{frame}
    Assim vemos que um ponto $D$ \pause pertence ao plano $\pi$ \pause se, e somente se, \pause $\vec{u}\times\vec{v} \perp \vec{AD}$. \pause

    Logo um plano $\pi$ é um conjunto de vetores perpendiculares a um dado vetor.\pause

    Fixe então um ponto $P_0$ em $\real^3$ \pause e $\vec{v}\ne \vec{0}$ um vetor. \pause Passando por $P_0$ \pause existe um único plano $\pi$ \pause perpendicular
    ao vetor $\vec{v}$. \pause Assim um ponto $P$ do espaço pertence ao plano $\pi$ \pause se, e somente se, \pause $\vec{P_0P} \perp \vec{v}$\pause, isto
    é,\pause
    \begin{equation}\label{equacaoplanovetorial}
        \inner{P_0P}{v} = 0.
    \end{equation}
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\vec{v} = (a,b,c)$ um vetor e \pause $P_0(x_0, y_0, z_0)$ \pause um ponto em $\real^3$. \pause Da equação
        \eqref{equacaoplanovetorial} \pause o ponto $P(x, y, z)$ em $\real^3$ \pause pertence ao plano definido por $P_0$ \pause e $\vec{v}$ \pause quando\pause
        \begin{equation}\label{equacaoplano}
            (x - x_0)a + (y - y_0)b + (z - z_0)c = 0.\pause
        \end{equation}

        A equação anterior é chamada de \textbf{equação cartesiana do plano} $\pi$. \pause O vetor $\vec{v}$ \pause é chamado de \textbf{vetor normal} \pause ao plano $\pi$.
    \end{definicao}
\end{frame}

\begin{frame}
    Agora, dada uma equação da forma
    \[
        ax + by + cz + d = 0,
    \]\pause
    onde $a$, $b$ e $c$ \pause não são simultaneamente nulos, \pause existe um plano $\pi$ representado por esta equação?
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\pi_1$ e $\pi_2$ \pause dois planos dados por\pause
        \begin{align*}
            \pi_1 : a_1x + b_1y + c_1z + d_1 = 0 \\
            \pi_2 : a_2x + b_2y + c_2z + d_2 = 0
        \end{align*}\pause
        onde os vetores normais de $\pi_1$ e $\pi_2$ \pause são, respectivamente,\pause
        \begin{align*}
            \vec{v_1} = (a_1, b_1, c_1) \\
            \vec{v_2} = (a_2, b_2, c_2).
        \end{align*}\pause
        Os planos $\pi_1$ e $\pi_2$ são \textbf{paralelos} \pause se, e somente se, \pause $\vec{v_1} \varparallel\vec{v_2}$.
    \end{definicao}
\end{frame}

\begin{frame}
    Seja $\vec{v}$ um vetor \pause e $A$ um ponto. \pause Sabemos que a reta $r$ \pause passando por $A$ \pause e com vetor diretor $\vec{v}$ \pause tem
    equação\pause
    \begin{equation}
        \vec{AP} = t\vec{v}
    \end{equation}\pause
    onde $t \in \real$ \pause e $P$ é um ponto qualquer de $r$. \pause Sendo $A$ um ponto em $\real^3$ \pause e $\vec{v}$ um vetor de $\real^3$ \pause
    temos\pause
    \begin{align*}
        A = (x_0, y_0, z_0) \\
        \vec{v} = (a, b, c).
    \end{align*}
\end{frame}

\begin{frame}
    Daí um ponto $P(x, y, z)$ \pause pertence à reta $r$ \pause se, e somente se,\pause
    \[
        \vec{AP} = (x - x_0, y - y_0, z - z_0) \pause = t(a, b, c).
    \]\pause
    Logo $P(x, y, z)$ \pause pertence à reta $r$, \pause se e só se, \pause $P(x,y,z)$ satisfaz as equações\pause
    \begin{equation}\label{equacoesparametricasretaespaco}
        \begin{cases}
            x = x_0 + ta \\
            y = y_0 + tb \\
            z = z_0 + tc
        \end{cases}; \quad t \in \real.
    \end{equation}\pause
    Tais equações são chamadas de \textbf{equações paramétricas} da reta $r$. \pause O vetor $\vec{v}$ é chamado de \textbf{vetor diretor} de $r$.
\end{frame}

\begin{frame}
    De forma análoga ao casa do plano, \pause se $\vec{v} = (a,b,c)$ \pause é tal que $abc \ne 0$, \pause então podemos reescrever
    \eqref{equacoesparametricasretaespaco} como\pause
    \[
        \dfrac{x - x_0}{a} = \dfrac{y - y_0}{b} = \dfrac{z - z_0}{c}
    \]\pause
    que são chamadas de \textbf{equações na forma simétrica} da reta $r$.
\end{frame}

\begin{frame}
    Considere duas retas $r$ e $s$ \pause de equações\pause
    \[
        r: \begin{cases}
            x = x_0 + ta \\
            y = y_0 + tb \\
            z = z_0 + tc
        \end{cases}; \quad \pause s: \begin{cases}
            x = x_1 + \alpha d \\
            y = y_1 + \alpha e \\
            z = z_1 + \alpha f
        \end{cases}; t,\ \alpha \in \real.
    \]\pause
    Os vetores diretores de $r$ e $s$ \pause são $\vec{u} = (a,b,c)$ \pause e $\vec{v} = (d,e,f)$, \pause respectivamente.
\end{frame}

\begin{frame}
    \begin{enumerate}
        \item[i)] Se $\vec{u}\varparallel\vec{v}$, \pause então $r$ e $s$ são paralelas \pause ou coincidentes.
    \end{enumerate}
\end{frame}

    Seja $P_1(x_1,y_1,z_1)$ um ponto e $\pi$ um plano de equação $ax + by + cz + d = 0$.

\begin{frame}
    Seja $r$ a reta que contém o ponto $P_1$ é é perpendicular ao plano $\pi$. Denote por $P_2(x_2,y_2,z_2)$ a interseção de $\pi$ com $r$. O ponto $P_2$ é chamado de \textbf{projeção ortogonal} de $P_1$ sobre $\pi$. A norma do vetor $\vec{P_1P_2}$ será a distância de $P_1$ a $\pi$, isto é,
    \[
        d(P_1,\pi) = \norm{\vec{P_1P_2}}.
    \]
\end{frame}

\begin{frame}
    Seja $\vec{u} = (a,b,c)$ o vetor normal de $\pi$. Como $\vec{u}\varparallel\vec{P_1P_2}$, existe $t \in \real$ tal que
    \[
        \vec{P_1P_2} = t\vec{u}.
    \]
    Logo
    \begin{equation}\label{equacaodistanciapontoplano}
        d(P_1,P_2) = \norm{t(a,b,c)} = |t|\sqrt{a^2 + b^2 + c^2}.
    \end{equation}
\end{frame}

\begin{frame}
    Por outro lado, se $P(x_2, y_2, z_2)$, então
    \[
        \vec{P_1P_2} = (x_2 - x_1, y_2 - y_1, z_2 - z_1) = t(a,b,c)
    \]
    e como $P_2 \in \pi$ devemos ter
    \[
        a(x_1 + ta) + b(y_1 + tb) + c(z_1 + tc) + d = 0
    \]
    isto é,
    \begin{equation}\label{equacaoauxiliardistanciapontoplano}
        t = -\dfrac{ax_1 + by_1 + cz_1 + d_1}{a^2 + b^2 + c^2}.
    \end{equation}
\end{frame}

\begin{frame}
    Substituindo \eqref{equacaoauxiliardistanciapontoplano} em \eqref{equacaodistanciapontoplano} obtemos
    \[
        d(P_1,\pi) = \left|\dfrac{ax_1 + by_1 + cz_1 + d_1}{a^2 + b^2 + c^2}\sqrt{a^2 + b^2 + c^2}\right|
    \]
    portanto
    \[
        d(P_1,\pi) = \dfrac{|ax_1 + by_1 + cz_1 + d_1|}{\sqrt{a^2 + b^2 + c^2}}.
    \]
\end{frame}

\begin{frame}
    Um outro modo de determinar essa distância é o seguinte: seja $P_0 \in \pi$. Então a distância de $P_1$ a $\pi$ é dada pela norma da projeção ortogonal do vetor $\vec{P_0P_1}$ na direção do vetor normal do plano $\vec{u} = (a,b,c)$. Assim
    \[
        d(P_1,\pi) = \norm{\proj_{\vec{u}}\vec{P_0P_1}} = \dfrac{|\langle\vec{P_0P_1},\vec{u}\rangle|}{\norm{\vec{u}}}.
    \]
\end{frame}

\begin{frame}
    Para determinar $d(P,r)$ primeiro construímos um plano $\pi$ passando por $P$ e com vetor normal paralelo ao vetor diretor de $r$. Assim
    \[
        d(P,r) = d(P,P_1)
    \]
    onde $P_1 \in \pi\cap r$.
\end{frame}

\begin{frame}
    Sejam $r$ e $s$ retas. Se $r \varparallel s$ então a distância entre $r$ e $s$ é dada pela distância entre um ponto de $r$ e a reta $s$, ou o contrário. Suponha então que $r$ e $s$ são concorrentes ou reversas. %Por um ponto $P$ de $s$ tracemos uma reta $s'$ paralela a $r$.

    Estas duas retas definem dois planos paralelos (que podem ser coincidentes, no caso em que as retas são concorrentes). Um plano é o que contém $r$ e é paralelo a $s$, que vamos chamar de $\pi_1$. O outro contém $s$ e é paralelo a $r$, que chamaremos de $\pi_2$.
\end{frame}

\begin{frame}
    Sejam $\vec{u}$ e $\vec{v}$ os vetores diretores de $r$ e $s$, respectivamente. Então o vetor $\vec{w} = \vec{u}\times\vec{v}$ é normal aos planos $\pi_1$ e $\pi_2$. Assim
    \[
        d(r,s) = d(\pi_1,\pi_2).
    \]
    Mas
    \[
        d(\pi_1,\pi_2) = d(\pi_1, Q) = \dfrac{|\langle\vec{PQ},\vec{w}\rangle|}{\norm{\vec{w}}} = \dfrac{|\langle\vec{PQ},\vec{u}\times\vec{v}\rangle|}{\norm{\vec{u}\times\vec{v}}}
    \]
    onde $P \in r$ e $Q \in s$. Portanto,
    \[
        d(r,s) = d(\pi_1, Q) = \dfrac{|\langle\vec{PQ},\vec{w}\rangle|}{\norm{\vec{w}}} = \dfrac{|\langle\vec{PQ},\vec{u}\times\vec{v}\rangle|}{\norm{\vec{u}\times\vec{v}}}.
    \]

    Se $r \cap s \ne \emptyset$, então $d(r,s) = 0$ pois $\vec{PQ}$, $\vec{u}$ e $\vec{v}$ estão no mesmo plano.
\end{frame}
