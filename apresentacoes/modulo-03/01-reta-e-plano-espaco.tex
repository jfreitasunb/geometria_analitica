\begin{frame}
    Da Geometria Plana, \pause sabemos que todo plano $\pi$ \pause contém pelo menos três pontos não colineares, \pause digamos $A$, \pause $B$ e $C$. \pause À
    partir desses pontos podemos construir os vetores:\pause
    \begin{align*}
        \vec{u} = \vec{AB} \\
        \vec{v} = \vec{AC}.
    \end{align*}
    Como $\vec{u}$ e $\vec{v}$ não são paralelos, \pause então podemos considerar o vetor $\vec{u}\times\vec{v}\ne \vec{0}$.\pause
\end{frame}

\begin{frame}
    \begin{figure}[!h]
        \centering
        \caption{Equação do plano}\label{definicao-plano}
        \hspace*{-3cm}
        \includegraphics{equacao-plano.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    Assim vemos que um ponto $D$ \pause pertence ao plano $\pi$ \pause se, e somente se, \pause $\vec{u}\times\vec{v} \perp \vec{AD}$. \pause

    Logo um plano $\pi$ é um conjunto de vetores perpendiculares a um dado vetor.\pause

    Fixe então um ponto $P_0$ em $\real^3$ \pause e $\vec{v}\ne \vec{0}$ um vetor. \pause Passando por $P_0$ \pause existe um único plano $\pi$ \pause perpendicular
    ao vetor $\vec{v}$. \pause Assim um ponto $P$ do espaço pertence ao plano $\pi$ \pause se, e somente se, \pause $\vec{P_0P} \perp \vec{v}$\pause, isto
    é,\pause
    \begin{equation}\label{equacaoplanovetorial}
        \inner{P_0P}{v} = 0.
    \end{equation}
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\vec{v} = (a,b,c)$ um vetor e \pause $P_0(x_0, y_0, z_0)$ \pause um ponto em $\real^3$. \pause Da equação
        \eqref{equacaoplanovetorial} \pause o ponto $P(x, y, z)$ em $\real^3$ \pause pertence ao plano definido por $P_0$ \pause e $\vec{v}$ \pause quando\pause
        \begin{equation}\label{equacaoplano}
            (x - x_0)a + (y - y_0)b + (z - z_0)c = 0.\pause
        \end{equation}

        A equação anterior é chamada de \textbf{equação cartesiana do plano} $\pi$. \pause O vetor $\vec{v}$ \pause é chamado de \textbf{vetor normal} \pause ao plano $\pi$.
    \end{definicao}
\end{frame}

\begin{frame}
    Agora, dada uma equação da forma
    \[
        ax + by + cz + d = 0,
    \]\pause
    onde $a$, $b$ e $c$ \pause não são simultaneamente nulos, \pause existe um plano $\pi$ representado por esta equação?
\end{frame}

\begin{frame}
    \begin{definicao}
        Sejam $\pi_1$ e $\pi_2$ \pause dois planos dados por\pause
        \begin{align*}
            \pi_1 : a_1x + b_1y + c_1z + d_1 = 0 \\
            \pi_2 : a_2x + b_2y + c_2z + d_2 = 0
        \end{align*}\pause
        onde os vetores normais de $\pi_1$ e $\pi_2$ \pause são, respectivamente,\pause
        \begin{align*}
            \vec{v_1} = (a_1, b_1, c_1) \\
            \vec{v_2} = (a_2, b_2, c_2).
        \end{align*}\pause
        Os planos $\pi_1$ e $\pi_2$ são \textbf{paralelos} \pause se, e somente se, \pause $\vec{v_1} \varparallel\vec{v_2}$.
    \end{definicao}
\end{frame}

\begin{frame}
    Seja $\vec{v}$ um vetor \pause e $A$ um ponto. \pause Sabemos que a reta $r$ \pause passando por $A$ \pause e com vetor diretor $\vec{v}$ \pause tem
    equação\pause
    \begin{equation}
        \vec{AP} = t\vec{v}
    \end{equation}\pause
    onde $t \in \real$ \pause e $P$ é um ponto qualquer de $r$. \pause Sendo $A$ um ponto em $\real^3$ \pause e $\vec{v}$ um vetor de $\real^3$ \pause
    temos\pause
    \begin{align*}
        A = (x_0, y_0, z_0) \\
        \vec{v} = (a, b, c).
    \end{align*}
\end{frame}

\begin{frame}
    Daí um ponto $P(x, y, z)$ \pause pertence à reta $r$ \pause se, e somente se,\pause
    \[
        \vec{AP} = (x - x_0, y - y_0, z - z_0) \pause = t(a, b, c).
    \]\pause
    Logo $P(x, y, z)$ \pause pertence à reta $r$, \pause se e só se, \pause $P(x,y,z)$ satisfaz as equações\pause
    \begin{equation}\label{equacoesparametricasretaespaco}
        \begin{cases}
            x = x_0 + ta \\
            y = y_0 + tb \\
            z = z_0 + tc
        \end{cases}; \quad t \in \real.
    \end{equation}\pause
    Tais equações são chamadas de \textbf{equações paramétricas} da reta $r$. \pause O vetor $\vec{v}$ é chamado de \textbf{vetor diretor} de $r$.
\end{frame}

\begin{frame}
    De forma análoga ao casa do plano, \pause se $\vec{v} = (a,b,c)$ \pause é tal que $abc \ne 0$, \pause então podemos reescrever
    \eqref{equacoesparametricasretaespaco} como
    \[
        \dfrac{x - x_0}{a} = \dfrac{y - y_0}{b} = \dfrac{z - z_0}{c}
    \]\pause
    que são chamadas de \textbf{equações na forma simétrica} \pause da reta $r$.
\end{frame}

\begin{frame}
    Considere duas retas $r$ e $s$ \pause de equações\pause
    \[
        r: \begin{cases}
            x = x_0 + \alpha a \\
            y = y_0 + \alpha b \\
            z = z_0 + \alpha c
        \end{cases}; \pause \quad  s: \begin{cases}
            x = x_1 + \beta d \\
            y = y_1 + \beta e \\
            z = z_1 + \beta f
        \end{cases};\ \alpha,\ \beta \in \real.\pause
    \]
    Os vetores diretores de $r$ e $s$ são \pause $\vec{u} = (a,b,c)$ \pause e $\vec{v} = (d,e,f)$, \pause respectivamente.
\end{frame}

\begin{frame}
    \begin{enumerate}
        \item[i)] Se $\vec{u}\varparallel\vec{v}$, \pause então $r$ e $s$ são paralelas \pause ou coincidentes.
    \end{enumerate}
\end{frame}

\begin{frame}
    \begin{enumerate}
        \item[ii)] Se os vetores diretores não são paralelos, \pause temos duas possibilidades. \pause Considere os pontos $P_0(x_0,y_0,z_0) \in r$ \pause e $P_1(x_1,y_1,z_1) \in s$.\pause
              \begin{enumerate}
                  \item[1)] Se $\vec{P_0P_1}$, \pause $\vec{u}$ \pause e $\vec{v}$ \pause estão no mesmo plano, \pause isto é, \pause $\langle\vec{P_0P_1}, \vec{u}\times\vec{v}\rangle = 0$ \pause então $r$ e $s$ são concorrentes.
              \end{enumerate}
    \end{enumerate}
\end{frame}

\begin{frame}

        \item[2)] Se $\vec{P_0P_1}$, \pause $\vec{u}$ \pause e $\vec{v}$ \pause estão planos diferentes, \pause isto é, \pause $\langle\vec{P_0P_1}, \vec{u}\times\vec{v}\rangle \ne 0$ \pause então $r$ e $s$ são chamadas de \textbf{retas reversas}.
    \end{enumerate}

\begin{frame}
    Seja $r$ a reta que contém o ponto $P_1$ \pause e é perpendicular ao plano $\pi$. \pause Denote por $P_2(x_2,y_2,z_2)$ \pause  a interseção de $\pi$ com $r$. \pause O ponto $P_2$ \pause é chamado de \textbf{projeção ortogonal} \pause de $P_1$ sobre $\pi$. \pause A norma do vetor $\vec{P_1P_2}$ \pause será a distância de $P_1$ a $\pi$, \pause isto é,\pause
    \[
        d(P_1,\pi) \pause = \norm{\vec{P_1P_2}} \pause = \dfrac{|ax_1 + by_1 + cz_1 + d_1|}{\sqrt{a^2 + b^2 + c^2}}.
    \]
\end{frame}

\begin{frame}
    Um outro modo de determinar essa distância é o seguinte: \pause seja $P_0 \in \pi$. \pause Então a distância de $P_1$ a $\pi$ \pause é dada pela norma da projeção ortogonal \pause do vetor $\vec{P_0P_1}$ \pause na direção do vetor normal do plano $\vec{u} = (a,b,c)$. \pause Assim
    \[
        d(P_1,\pi) \pause = \norm{\proj_{\vec{u}}\vec{P_0P_1}} \pause = \dfrac{|\langle\vec{P_0P_1},\vec{u}\rangle|}{\norm{\vec{u}}}.
    \]
\end{frame}

\begin{frame}
    Dados um ponto $P$ e uma reta $r$ como determinar a distância de $P$ a $r$?
\end{frame}

\begin{frame}
    Para determinar $d(P,r)$ \pause primeiro construímos um plano $\pi$ \pause passando por $P$ \pause e com vetor normal \pause paralelo ao vetor diretor de $r$.
\end{frame}

\begin{frame}
    Seja $P$ um ponto e $r$ uma reta.
    \begin{figure}[h]
        \centering

\begin{frame}
    Assim
    \[
    d(P,r) = d(P,P_1)
    \]
    onde $P_1 \in \pi\cap r$.
\end{frame}

\begin{frame}
    Sejam $r$ e $s$ retas. \pause Se $r \varparallel s$ \pause então a distância entre $r$ e $s$ \pause é dada pela distância entre um ponto de $r$ \pause e a reta $s$, \pause ou o contrário. \pause Suponha então que $r$ e $s$ são concorrentes ou reversas. \pause

    \vspace{.5cm}

    Se $r \cap s \ne \emptyset$, então $d(r,s) = 0$.\pause

    \vspace{.5cm}

    Vamos considerar então o caso em que $r$ e $s$ são retas reversas.
\end{frame}

\begin{frame}
    Estas duas retas definem dois planos paralelos. \pause

    \vspace{.5cm}

    Um plano é o que contém $r$ e é paralelo a $s$, \pause que vamos chamar de $\pi_1$. \pause

    \vspace{.5cm}

    O outro contém $s$ e é paralelo a $r$, \pause que chamaremos de $\pi_2$.
\end{frame}

    \begin{figure}[!h]

\begin{frame}
    Sejam $\vec{u}$ \pause e $\vec{v}$ \pause os vetores diretores \pause de $r$ e $s$, \pause respectivamente. Então o vetor $\vec{u}\times\vec{v}$ \pause é normal aos planos $\pi_1$ \pause e $\pi_2$. \pause Assim
    \[
        d(r,s) = d(\pi_1,\pi_2).
    \]\pause
    Mas
    \[
        d(\pi_1,\pi_2) = \pause \dfrac{|\langle\vec{PQ},\vec{u}\times\vec{v}\rangle|}{\norm{\vec{u}\times\vec{v}}}
    \]\pause
    onde $P \in r$ \pause e $Q \in s$. \pause Portanto,
    \[
        d(r,s) \pause = \dfrac{|\langle\vec{PQ},\vec{u}\times\vec{v}\rangle|}{\norm{\vec{u}\times\vec{v}}}.
    \]
\end{frame}
